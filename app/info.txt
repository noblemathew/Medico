def col_number(ref):
    if isinstance(ref, int):
        return ref
    n = 0
    for ch in str(ref).strip().upper():
        if not ch.isalpha():
            raise ValueError(f"Bad column reference: {ref!r}")
        n = n * 26 + (ord(ch) - 64)
    return n


def col_letter(n):
    out = ""
    while n > 0:
        n, rem = divmod(n - 1, 26)
        out = chr(65 + rem) + out
    return out

def find_file(prefix, folder=None, must_contain=None,
              must_not_contain=None, require_date=True):
    folder = Path(folder).resolve() if folder else Path(FOLDER).resolve()
    key = prefix.strip().lower()
    need = [k.lower() for k in (must_contain or [])]
    ban = [k.lower() for k in (must_not_contain or [])]
    hits = []

    for f in folder.iterdir():
        if not f.is_file() or f.name.startswith("~$"):
            continue
        if f.suffix.lower() not in (".xlsx", ".xlsm", ".xls"):
            continue

        stem = f.stem.strip().lower()

        if not stem.startswith(key):
            continue
        if any(k not in stem for k in need):
            continue
        if any(k in stem for k in ban):
            continue

        if require_date:
            m = re.search(r"\d{6}", f.stem)
            if not m:
                continue
            try:
                d = datetime.strptime(m.group(0), FILENAME_DATE_FORMAT)
            except ValueError:
                continue
            hits.append((d, f))
        else:
            hits.append((None, f))

    if not hits:
        present = [
            f.name for f in folder.iterdir()
            if f.is_file()
            and f.suffix.lower() in (".xlsx", ".xlsm", ".xls")
            and not f.name.startswith("~$")
        ]
        raise FileNotFoundError(
            f"No Excel file starting with '{prefix}' in {folder}\n"
            f"    Files present: {present or 'none'}"
        )

    if require_date:
        distinct = {
            re.sub(r"\d{6}", "", f.stem).strip().lower() for _, f in hits
        }
        if len(distinct) > 1:
            raise ValueError(
                f"'{prefix}' matches {len(distinct)} different files: "
                f"{sorted(distinct)}"
            )
        hits.sort(key=lambda x: x[0])
        return hits[-1][1]

    if len(hits) > 1:
        raise ValueError(
            f"Multiple Excel files found starting with '{prefix}': "
            f"{[f.name for _, f in hits]}"
        )

    return hits[0][1]


def find_webi_file(report, log=print):
    custom = (report.get("webi_name") or "").strip()
    prefix = custom or (_setting("WEBI_PREFIX", "Webi_") + report["name"])

    guards = {}
    if not custom:
        guards = {
            "must_contain": report.get("must_contain"),
            "must_not_contain": report.get("must_not_contain"),
        }

    try:
        return find_file(prefix, require_date=True, **guards)
    except FileNotFoundError:
        return find_file(prefix, require_date=False, **guards)


def next_target_day(d):
    ahead = (TARGET_WEEKDAY - d.weekday()) % 7
    return d + timedelta(days=ahead or 7)


def next_month_date(d):
    import calendar
    y, m = (d.year + 1, 1) if d.month == 12 else (d.year, d.month + 1)
    last = calendar.monthrange(y, m)[1]
    if _setting("MONTHLY_DATE_RULE", "month_end") == "same_day":
        return d.replace(year=y, month=m, day=min(d.day, last))
    return d.replace(year=y, month=m, day=last)


def build_new_filename(path, log=print, mode="weekly"):
    p = Path(path)
    stem, ext = p.stem, p.suffix

    m = None
    for cand in reversed(list(re.finditer(r"\d{6}", stem))):
        try:
            datetime.strptime(cand.group(0), FILENAME_DATE_FORMAT)
            m = cand
            break
        except ValueError:
            continue

    if not m:
        raise ValueError(f"No valid MMDDYY date in '{p.name}'")

    old = datetime.strptime(m.group(0), FILENAME_DATE_FORMAT)
    new = next_month_date(old) if mode == "monthly" else next_target_day(old)
    log(f"    date {old:%d-%b-%Y} ({old:%a}) -> {new:%d-%b-%Y} ({new:%a})")

    new_stem = (stem[:m.start()]
                + new.strftime(FILENAME_DATE_FORMAT)
                + stem[m.end():])
    return str(p.parent / (new_stem + ext))


def report_end_date(paths):
    for p in paths:
        for cand in reversed(list(re.finditer(r"\d{6}", Path(p).stem))):
            try:
                d = datetime.strptime(cand.group(0), FILENAME_DATE_FORMAT)
                return d.strftime("%d %b %Y")
            except ValueError:
                continue
    return ""


def last_row(sheet):
    ur = sheet.api.UsedRange
    return ur.Row + ur.Rows.Count - 1


def last_col(sheet):
    ur = sheet.api.UsedRange
    return ur.Column + ur.Columns.Count - 1


def find_marker_row(sheet, start_row, marker):
    end_r, end_c = last_row(sheet), last_col(sheet)
    if end_r < start_row:
        return None

    needle = marker.strip()

    try:
        escaped = (needle.replace("~", "~~")
                         .replace("*", "~*")
                         .replace("?", "~?"))
        rng = sheet.api.Range(sheet.api.Cells(start_row, 1),
                              sheet.api.Cells(end_r, end_c))
        last_cell = rng.Cells(rng.Rows.Count, rng.Columns.Count)
        hit = rng.Find(escaped, last_cell, XL_VALUES, XL_PART,
                       XL_BY_ROWS, XL_NEXT, False)
        if hit is not None:
            return hit.Row
        return None
    except Exception:
        pass

    low = needle.lower()
    best = None
    for c in range(1, end_c + 1):
        vals = sheet.range((start_row, c), (end_r, c)).value
        if not isinstance(vals, list):
            vals = [vals]
        for i, v in enumerate(vals):
            if v is not None and low in str(v).lower():
                r = start_row + i
                if best is None or r < best:
                    best = r
                break
    return best


def find_disclaimer_top(sheet, marker_row, data_start, configured_above=0,
                        max_scan=15):
    end_c = last_col(sheet)
    r = marker_row
    for _ in range(max_scan):
        above = r - 1
        if above <= data_start:
            break
        vals = sheet.range((above, 1), (above, end_c)).value
        if not isinstance(vals, list):
            vals = [vals]
        if all(v is None or (isinstance(v, str) and not v.strip())
               for v in vals):
            return r
        r = above

    return marker_row - configured_above


def go_to_a1(sheet):
    try:
        sheet.api.Activate()
        sheet.book.app.api.Goto(sheet.api.Range("A1"), True)
    except Exception:
        try:
            sheet.api.Range("A1").Select()
        except Exception:
            pass


def find_webi_last_row(src_ws, key_col, src_start):
    return src_ws.api.Cells(src_ws.api.Rows.Count, key_col).End(XL_UP).Row


def paste_values_in_chunks(source_ws, destination_ws,
                           source_start_row, source_end_row,
                           source_first_col, source_last_col,
                           destination_start_row, destination_first_col,
                           chunk_size=10000, log=print):
    total_rows = source_end_row - source_start_row + 1
    width = source_last_col - source_first_col + 1
    dst_col2 = destination_first_col + width - 1

    copied = 0
    while copied < total_rows:
        rows_this_chunk = min(chunk_size, total_rows - copied)

        src_row1 = source_start_row + copied
        src_row2 = src_row1 + rows_this_chunk - 1

        dst_row1 = destination_start_row + copied
        dst_row2 = dst_row1 + rows_this_chunk - 1

        source_range = source_ws.range((src_row1, source_first_col),
                                       (src_row2, source_last_col))
        destination_range = destination_ws.range((dst_row1, destination_first_col),
                                                 (dst_row2, dst_col2))

        if source_range.rows.count != destination_range.rows.count:
            raise ValueError(f"Row mismatch: source {src_row1}:{src_row2}, "
                             f"destination {dst_row1}:{dst_row2}")
        if source_range.columns.count != destination_range.columns.count:
            raise ValueError(f"Column mismatch in chunk starting at "
                             f"source row {src_row1}")

        destination_range.api.Value2 = source_range.api.Value2

        copied += rows_this_chunk
        log(f"    copied {copied:,} / {total_rows:,} rows")


def _as_number(v):
    if isinstance(v, bool):
        return None
    if isinstance(v, (int, float)):
        return float(v)
    if isinstance(v, str):
        t = v.strip().replace(",", "")
        try:
            return float(t)
        except ValueError:
            return None
    return None


def _qc_compare(a, b):
    blank_a = a is None or (isinstance(a, str) and a.strip() == "")
    blank_b = b is None or (isinstance(b, str) and b.strip() == "")
    if blank_a and blank_b:
        return "same"
    if blank_a != blank_b:
        return "diff"
    if a == b:
        return "same"

    na, nb = _as_number(a), _as_number(b)
    if na is not None and nb is not None:
        if abs(na - nb) <= 1e-9 * max(1.0, abs(na), abs(nb)):
            return "type" if type(a) is not type(b) else "same"
        return "diff"

    if isinstance(a, str) and isinstance(b, str) and a.strip() == b.strip():
        return "same"
    return "diff"


def _grid(values):
    if values is None or not isinstance(values, (list, tuple)):
        return [[values]]
    if values and not isinstance(values[0], (list, tuple)):
        return [list(values)]
    return [list(r) for r in values]


def qc_sheet(ws, src_ws, start, c1, c2, src_start, src_last_row, src_c1,
             src_c2, disclaimer_text, block_offset, blank_rows, log=print,
             chunk_size=10000, max_report=10):
    name = ws.name
    rows = src_last_row - src_start + 1
    last = start + rows - 1
    problems = []

    diffs, type_diffs, shown = 0, 0, 0
    done = 0
    while done < rows:
        n = min(chunk_size, rows - done)
        a = _grid(ws.range((start + done, c1),
                           (start + done + n - 1, c2)).api.Value2)
        b = _grid(src_ws.range((src_start + done, src_c1),
                               (src_start + done + n - 1, src_c2)).api.Value2)

        for i, (ra, rb) in enumerate(zip(a, b)):
            for j, (va, vb) in enumerate(zip(ra, rb)):
                res = _qc_compare(va, vb)
                if res == "same":
                    continue
                if res == "type":
                    type_diffs += 1
                    continue
                diffs += 1
                if shown < max_report:
                    shown += 1
                    log(f"      diff {col_letter(c1 + j)}{start + done + i}: "
                        f"report {va!r} | webi {vb!r} "
                        f"(webi {col_letter(src_c1 + j)}"
                        f"{src_start + done + i})")
        done += n

    if diffs:
        problems.append(f"{diffs:,} cell(s) differ")
    if type_diffs:
        log(f"    QC note: {type_diffs:,} cell(s) match in value but one "
            "side is stored as text - check the column number format.")

    below = _grid(ws.range((last + 1, c1), (last + 1, c2)).api.Value2)[0]
    if any(v not in (None, "") for v in below):
        problems.append(f"row {last + 1} (should be blank) has data: {below}")

    marker = find_marker_row(ws, last + 1, disclaimer_text)
    if marker is None:
        problems.append("disclaimer not found below the data")
    else:
        top = marker - block_offset
        expected = last + blank_rows + 1
        if top != expected:
            problems.append(f"disclaimer starts at row {top}, "
                            f"expected row {expected}")

    for label, r_rep, r_src in (("first", start, src_start),
                                ("last ", last, src_last_row)):
        rep = _grid(ws.range((r_rep, c1), (r_rep, c2)).api.Value2)[0]
        src = _grid(src_ws.range((r_src, src_c1), (r_src, src_c2)).api.Value2)[0]
        log(f"    QC {label} row  report {r_rep}: {rep}")
        log(f"    QC {label} row  webi   {r_src}: {src}")

    cells = rows * (c2 - c1 + 1)
    if problems:
        log(f"    QC FAILED on '{name}': " + "; ".join(problems))
        return False

    log(f"    QC PASSED on '{name}': {rows:,} rows, {cells:,} cells match, "
        f"disclaimer {blank_rows} row(s) below data")
    return True


def process_sheet(wb, webi_wb, cfg, log=print):
    name = cfg["old_sheet"]
    blank_rows = BLANK_ROWS_BEFORE_DISCLAIMER
    above_marker = cfg.get("disclaimer_rows_above_marker", 0) or 0

    sheet_names = [s.name for s in wb.sheets]
    if name not in sheet_names:
        raise ValueError(f"Tab '{name}' not found. Tabs: {sheet_names}")

    ws = wb.sheets[name]
    start = cfg["old_data_start_row"]
    c1 = col_number(cfg["old_first_col"])
    c2 = col_number(cfg["old_last_col"])
    width = c2 - c1 + 1

    log(f"  Tab '{name}'  <-  WebI '{cfg['webi_sheet']}'")

    webi_sheet_name = cfg["webi_sheet"]
    available = [s.name for s in webi_wb.sheets]
    if webi_sheet_name not in available:
        raise ValueError(f"WebI tab '{webi_sheet_name}' not found. "
                         f"Tabs: {available}")

    src_ws = webi_wb.sheets[webi_sheet_name]
    src_start = cfg["webi_data_start_row"]
    src_c1 = col_number(cfg["webi_first_col"])
    src_c2 = col_number(cfg["webi_last_col"])
    src_width = src_c2 - src_c1 + 1

    if src_width != width:
        raise ValueError(
            f"Column count mismatch on '{name}': WebI gives {src_width} "
            f"({cfg['webi_first_col']}-{cfg['webi_last_col']}) but report "
            f"expects {width} ({cfg['old_first_col']}-{cfg['old_last_col']})."
        )

    key_col = col_number(cfg.get("webi_key_col", cfg["webi_first_col"]))
    src_last_row = find_webi_last_row(src_ws, key_col, src_start)

    if src_last_row < src_start:
        raise ValueError(f"No data found in WebI tab '{webi_sheet_name}' - "
                         "stopping so the old data isn't wiped.")

    new_count = src_last_row - src_start + 1

    check_start = max(src_start, src_last_row - 2)
    log(f"    WebI {col_letter(src_c1)}{src_start}:"
        f"{col_letter(src_c2)}{src_last_row}  ({new_count:,} rows)")
    try:
        tail = src_ws.range((check_start, src_c1),
                            (src_last_row, src_c2)).options(ndim=2).value
        for i, row in enumerate(tail, start=check_start):
            log(f"      WebI row {i}: {row}")
    except Exception:
        pass

    marker = find_marker_row(ws, start, cfg["disclaimer_marker_text"])
    if marker is None:
        raise ValueError(f"Disclaimer not found on '{name}'. "
                         "Check disclaimer_marker_text.")

    disc_start = find_disclaimer_top(ws, marker, start, above_marker)
    block_offset = marker - disc_start
    if disc_start <= start:
        raise ValueError(f"Disclaimer row {disc_start} is at/above data "
                         f"start row {start} on '{name}'.")

    old_count = max(0, disc_start - blank_rows - start)

    new_last_data = start + new_count - 1
    required_disc_start = new_last_data + blank_rows + 1

    log(f"    old rows: {old_count:,}   new rows: {new_count:,}")
    log(f"    disclaimer: row {disc_start:,} -> row {required_disc_start:,}")

    row_difference = required_disc_start - disc_start

    if row_difference > 0:
        ws.api.Rows(f"{disc_start}:{disc_start + row_difference - 1}").Insert()
        log(f"    inserted {row_difference:,} row(s)")
    elif row_difference < 0:
        n = -row_difference
        ws.api.Rows(f"{disc_start - n}:{disc_start - 1}").Delete()
        log(f"    deleted {n:,} row(s)")

    ws.range((start, c1), (required_disc_start - 1, c2)).clear_contents()

    paste_values_in_chunks(
        source_ws=src_ws,
        destination_ws=ws,
        source_start_row=src_start,
        source_end_row=src_last_row,
        source_first_col=src_c1,
        source_last_col=src_c2,
        destination_start_row=start,
        destination_first_col=c1,
        chunk_size=_setting("CHUNK_SIZE", 10000),
        log=log,
    )

    marker_after = find_marker_row(ws, new_last_data + 1,
                                   cfg["disclaimer_marker_text"])
    if marker_after is None:
        raise ValueError(f"Disclaimer lost after paste on '{name}'.")

    disc_top = marker_after - block_offset

    if disc_top - 1 >= new_last_data + 1:
        ws.range((new_last_data + 1, c1), (disc_top - 1, c2)).clear_contents()

    gap = disc_top - new_last_data - 1
    if gap < blank_rows:
        need = blank_rows - gap
        ws.api.Rows(f"{disc_top}:{disc_top + need - 1}").Insert()
        log(f"    fixed gap: inserted {need} row(s)")
    elif gap > blank_rows:
        extra = gap - blank_rows
        ws.api.Rows(f"{new_last_data + 1}:{new_last_data + extra}").Delete()
        log(f"    fixed gap: deleted {extra} row(s)")

    log(f"    final: data {col_letter(c1)}{start}:{col_letter(c2)}"
        f"{new_last_data}, disclaimer at row "
        f"{new_last_data + blank_rows + 1:,}")

    if _setting("QC_ENABLED", True):
        passed = qc_sheet(
            ws, src_ws,
            start=start, c1=c1, c2=c2,
            src_start=src_start, src_last_row=src_last_row,
            src_c1=src_c1, src_c2=src_c2,
            disclaimer_text=cfg["disclaimer_marker_text"],
            block_offset=block_offset,
            blank_rows=blank_rows,
            log=log,
        )
        if not passed and _setting("QC_BLOCKS_SAVE", True):
            raise ValueError(f"QC failed on '{name}' - report NOT saved. "
                             "See the diff lines above.")

    go_to_a1(ws)


def rename_when_free(src, dst, log=print, attempts=10):
    for i in range(attempts):
        try:
            Path(src).rename(dst)
            return
        except PermissionError:
            time.sleep(0.5)
    raise PermissionError(f"Could not rename '{Path(src).name}' - "
                          "is it open in Excel?")


def process_report(report, log=print, mode="weekly"):
    name = report["name"]
    log(f"\n=== {name} ===")

    old_path = find_file(name,
                         must_contain=report.get("must_contain"),
                         must_not_contain=report.get("must_not_contain"),
                         require_date=True)
    webi_path = find_webi_file(report, log=log)

    log(f"    report: {old_path.name}")
    log(f"    webi  : {webi_path.name}")

    rename = _setting("RENAME_WITH_NEXT_DATE", True)
    new_path = (build_new_filename(old_path, log=log, mode=mode)
                if rename else str(old_path))

    if rename and os.path.exists(new_path):
        raise ValueError(f"A file named {os.path.basename(new_path)} already "
                         "exists - not overwriting it.")

    app = xw.App(visible=EXCEL_VISIBLE, add_book=False)
    app.display_alerts = False
    app.screen_updating = False

    wb = None
    webi_wb = None
    saved = False
    try:
        wb = app.books.open(str(old_path.resolve()), update_links=False)
        webi_wb = app.books.open(str(Path(webi_path).resolve()),
                                 update_links=False, read_only=True)

        app.api.Calculation = XL_CALC_MANUAL
        app.api.EnableEvents = False

        for cfg in report["sheets"]:
            process_sheet(wb, webi_wb, cfg, log=log)

        app.api.Calculation = XL_CALC_AUTOMATIC
        app.api.EnableEvents = True

        first = wb.sheets[report["sheets"][0]["old_sheet"]]
        first.activate()
        go_to_a1(first)

        wb.save()
        saved = True

    finally:
        try:
            app.api.Calculation = XL_CALC_AUTOMATIC
            app.api.EnableEvents = True
            app.screen_updating = True
        except Exception:
            pass
        if webi_wb is not None:
            try:
                webi_wb.close()
            except Exception:
                pass
        if wb is not None:
            try:
                wb.close()
            except Exception:
                pass
        app.quit()

    if saved and rename:
        rename_when_free(old_path, new_path, log=log)
        log(f"    updated and renamed: {old_path.name} -> "
            f"{os.path.basename(new_path)}")
    elif saved:
        log(f"    updated: {old_path.name}")

    GENERATED.append(new_path)


def _row_is_blank(vals):
    return all(v is None or (isinstance(v, str) and not v.strip())
               for v in vals)


def _report_last_data_row(ws, start, c1, c2, disc_top):
    r = disc_top - 1
    while r >= start:
        if not _row_is_blank(_grid(ws.range((r, c1), (r, c2)).api.Value2)[0]):
            return r
        r -= 1
    return start - 1


def qc_tab_readonly(wb, webi_wb, cfg, log=print, max_report=10):
    name = cfg["old_sheet"]
    blank_rows = BLANK_ROWS_BEFORE_DISCLAIMER
    problems = []

    log(f"  QC tab '{name}'  vs  WebI '{cfg['webi_sheet']}'")

    if name not in [x.name for x in wb.sheets]:
        log(f"    tab '{name}' not found in report")
        log("    QC FAILED")
        return False
    if cfg["webi_sheet"] not in [x.name for x in webi_wb.sheets]:
        log(f"    tab '{cfg['webi_sheet']}' not found in WebI")
        log("    QC FAILED")
        return False

    ws = wb.sheets[name]
    src = webi_wb.sheets[cfg["webi_sheet"]]

    start = cfg["old_data_start_row"]
    c1 = col_number(cfg["old_first_col"])
    c2 = col_number(cfg["old_last_col"])
    s_start = cfg["webi_data_start_row"]
    s_c1 = col_number(cfg["webi_first_col"])
    s_c2 = col_number(cfg["webi_last_col"])

    if (c2 - c1) != (s_c2 - s_c1):
        log(f"    column count differs - report {c2 - c1 + 1}, "
            f"WebI {s_c2 - s_c1 + 1}")
        log("    QC FAILED")
        return False
    width = c2 - c1 + 1

    key_col = col_number(cfg.get("webi_key_col", cfg["webi_first_col"]))
    s_last = find_webi_last_row(src, key_col, s_start)
    webi_rows = max(0, s_last - s_start + 1)

    above = cfg.get("disclaimer_rows_above_marker", 0) or 0
    marker = find_marker_row(ws, start, cfg["disclaimer_marker_text"])
    if marker is None:
        problems.append("disclaimer not found")
        r_last = ws.api.Cells(ws.api.Rows.Count, c1).End(XL_UP).Row
    else:
        disc_top = find_disclaimer_top(ws, marker, start, above)
        r_last = _report_last_data_row(ws, start, c1, c2, disc_top)
        gap = disc_top - r_last - 1
        if gap != blank_rows:
            problems.append(f"gap before disclaimer is {gap} row(s), "
                            f"expected {blank_rows}")
    report_rows = max(0, r_last - start + 1)

    log(f"    rows: report {report_rows:,} | WebI {webi_rows:,}")
    if report_rows != webi_rows:
        problems.append(f"row count differs by {report_rows - webi_rows:+,}")

    n = min(report_rows, webi_rows)
    chunk = _setting("CHUNK_SIZE", 10000)
    diffs = type_diffs = shown = 0

    for off in range(0, n, chunk):
        k = min(chunk, n - off)
        a = _grid(ws.range((start + off, c1),
                           (start + off + k - 1, c2)).api.Value2)
        b = _grid(src.range((s_start + off, s_c1),
                            (s_start + off + k - 1, s_c2)).api.Value2)
        for i in range(k):
            for j in range(width):
                res = _qc_compare(a[i][j], b[i][j])
                if res == "same":
                    continue
                if res == "type":
                    type_diffs += 1
                    continue
                diffs += 1
                if shown < max_report:
                    shown += 1
                    log(f"      diff {col_letter(c1 + j)}{start + off + i}: "
                        f"report {a[i][j]!r} | WebI "
                        f"{col_letter(s_c1 + j)}{s_start + off + i} "
                        f"{b[i][j]!r}")

    log(f"    cells compared: {n * width:,}   differences: {diffs:,}")
    if diffs:
        problems.append(f"{diffs:,} cell(s) differ")
    if type_diffs:
        log(f"    note: {type_diffs:,} cell(s) equal in value but one side "
            "is stored as text")

    if n:
        for label, r_rep, r_src in (("first", start, s_start),
                                    ("last ", start + n - 1, s_start + n - 1)):
            rep = _grid(ws.range((r_rep, c1), (r_rep, c2)).api.Value2)[0]
            web = _grid(src.range((r_src, s_c1), (r_src, s_c2)).api.Value2)[0]
            log(f"    {label} row  report {r_rep}: {list(rep)}")
            log(f"    {label} row  webi   {r_src}: {list(web)}")

    if problems:
        for p_ in problems:
            log(f"    - {p_}")
        log("    QC FAILED")
        return False

    log("    QC PASSED")
    return True


def qc_report(report, log=print):
    name = report["name"]
    log(f"\n=== QC: {name} ===")

    rep_path = find_file(name,
                         must_contain=report.get("must_contain"),
                         must_not_contain=report.get("must_not_contain"),
                         require_date=True)
    webi_path = find_webi_file(report, log=log)
    log(f"    report: {rep_path.name}   (read-only)")
    log(f"    webi  : {webi_path.name}   (read-only)")

    app = xw.App(visible=False, add_book=False)
    app.display_alerts = False
    app.screen_updating = False
    try:
        app.api.EnableEvents = False
    except Exception:
        pass

    wb = webi_wb = None
    try:
        wb = app.books.open(str(rep_path.resolve()),
                            update_links=False, read_only=True)
        webi_wb = app.books.open(str(Path(webi_path).resolve()),
                                 update_links=False, read_only=True)
        results = [qc_tab_readonly(wb, webi_wb, cfg, log=log)
                   for cfg in report["sheets"]]
        return all(results)
    finally:
        for b in (webi_wb, wb):
            if b is not None:
                try:
                    b.close()
                except Exception:
                    pass
        app.quit()


def make_draft_email(folder, report_date="", to="", cc="", log=print):
    folder = str(Path(folder).resolve())
    today = datetime.now().strftime("%d %b %Y")

    subject_tpl = _setting("EMAIL_SUBJECT",
                           "Weekly Reports - Week Ending {report_date}")
    body_tpl = _setting("EMAIL_BODY", (
        "Hi,\n\n"
        "The weekly reports have been refreshed. You can find them here:\n\n"
        "{folder}\n\n"
        "Regards,\n"
    ))

    fields = {"date": today, "report_date": report_date or today}
    subject = subject_tpl.format(folder=folder, **fields)

    try:
        import win32com.client

        outlook = win32com.client.Dispatch("Outlook.Application")
        mail = outlook.CreateItem(0)
        mail.To = to
        if cc:
            mail.CC = cc
        mail.Subject = subject

        uri = "file:///" + folder.replace("\\", "/").replace(" ", "%20")
        link = f'<a href="{uri}">{html.escape(folder)}</a>'

        text = body_tpl.format(folder="\x00FOLDER\x00", **fields)
        safe = html.escape(text).replace("\n", "<br>")
        mail.HTMLBody = (
            '<div style="font-family:Calibri,sans-serif;font-size:11pt">'
            + safe.replace("\x00FOLDER\x00", link) + "</div>"
        )
        mail.Display(False)
        log("    draft opened in Outlook")
        return "outlook"
    except Exception as e:
        log(f"    Outlook not available ({type(e).__name__}) - using mailto")

    params = {"subject": subject, "body": body_tpl.format(folder=folder, **fields)}
    if cc:
        params["cc"] = cc
    url = ("mailto:" + urllib.parse.quote(to) + "?"
           + urllib.parse.urlencode(params, quote_via=urllib.parse.quote))
    webbrowser.open(url)
    log("    draft opened in your default mail client")
    return "mailto"


def _enable_hidpi():
    if not sys.platform.startswith("win"):
        return 1.0
    try:
        import ctypes
        try:
            ctypes.windll.shcore.SetProcessDpiAwareness(1)
        except Exception:
            ctypes.windll.user32.SetProcessDPIAware()
        dc = ctypes.windll.user32.GetDC(0)
        dpi = ctypes.windll.gdi32.GetDeviceCaps(dc, 88)
        ctypes.windll.user32.ReleaseDC(0, dc)
        return (dpi or 96) / 96.0
    except Exception:
        return 1.0

def _mix(c1, c2, t):
    a = [int(c1[i:i + 2], 16) for i in (1, 3, 5)]
    b = [int(c2[i:i + 2], 16) for i in (1, 3, 5)]
    return "#%02x%02x%02x" % tuple(round(x + (y - x) * t) for x, y in zip(a, b))


def _ease(t):
    return t * t * (3 - 2 * t)


def _round_rect(cv, x1, y1, x2, y2, r, **kw):
    r = max(0, min(r, (x2 - x1) / 2, (y2 - y1) / 2))
    pts = [x1 + r, y1, x2 - r, y1, x2, y1, x2, y1 + r,
           x2, y2 - r, x2, y2, x2 - r, y2, x1 + r, y2,
           x1, y2, x1, y2 - r, x1, y1 + r, x1, y1]
    return cv.create_polygon(pts, smooth=True, **kw)


class RoundButton(tk.Canvas):

    def __init__(self, parent, text, command, *, width, height, px,
                 bg, fg, hover_bg, hover_fg=None, border=None,
                 parent_bg, font, radius=10):
        super().__init__(parent, width=width, height=height, bg=parent_bg,
                         highlightthickness=0, bd=0, cursor="hand2")
        self._cmd = command
        self._text = text
        self._cw, self._ch, self._r = width, height, radius
        self._colors = dict(bg=bg, fg=fg, hbg=hover_bg,
                            hfg=hover_fg or fg, border=border)
        self._font = font
        self._parent_bg = parent_bg
        self._t = 0.0
        self._target = 0.0
        self._job = None
        self._enabled = True
        self._draw()
        self.bind("<Enter>", lambda e: self._chover(1.0))
        self.bind("<Leave>", lambda e: self._chover(0.0))
        self.bind("<ButtonRelease-1>", self._click)

    def _draw(self):
        self.delete("all")
        c = self._colors
        if self._enabled:
            fill = _mix(c["bg"], c["hbg"], _ease(self._t))
            fg = _mix(c["fg"], c["hfg"], _ease(self._t))
            outline = c["border"] or fill
        else:
            pb = self._parent_bg
            light = sum(int(pb[i:i + 2], 16) for i in (1, 3, 5)) > 600
            if light:
                fill, fg = "#eceef2", "#a8afbb"
                outline = "#e1e4ea"
            else:
                fill = _mix(pb, WHITE, 0.10)
                fg = _mix(pb, WHITE, 0.42)
                outline = _mix(pb, WHITE, 0.22) if c["border"] else fill
        _round_rect(self, 1, 1, self._cw - 1, self._ch - 1, self._r,
                    fill=fill, outline=outline, width=1)
        self.create_text(self._cw / 2, self._ch / 2, text=self._text,
                         fill=fg, font=self._font)

    def _hover(self, target):
        if not self._enabled:
            return
        self._target = target
        if self._job is None:
            self._animate()

    def _animate(self):
        step = 0.16 if self._target > self._t else -0.16
        self._t = min(1.0, max(0.0, self._t + step))
        self._draw()
        if abs(self._t - self._target) > 1e-6:
            self._job = self.after(16, self._animate)
        else:
            self._job = None

    def _click(self, e):
        if self._enabled and 0 <= e.x <= self._cw and 0 <= e.y <= self._ch:
            self._cmd()

    def set_text(self, text):
        self._text = text
        self._draw()

    def set_enabled(self, on):
        self._enabled = on
        self.config(cursor="hand2" if on else "arrow")
        if not on:
            self._t = self._target = 0.0
        self._draw()


class Segmented(tk.Canvas):

    def __init__(self, parent, options, command, *, width, height, px,
                 parent_bg, font, font_bold, track="#7a0819", pill="#ffffff",
                 on_fg="#c8102e", off_fg="#f4c9d0"):
        super().__init__(parent, width=width, height=height, bg=parent_bg,
                         highlightthickness=0, bd=0, cursor="hand2")
        self._track, self._pill = track, pill
        self._on_fg, self._off_fg = on_fg, off_fg
        self._opts = options
        self._cmd = command
        self._cw, self._ch, self._px = width, height, px
        self._font, self._font_b = font, font_bold
        self._index = 0
        self._pos = 0.0
        self._job = None
        self._enabled = True
        self._draw()
        self.bind("<ButtonRelease-1>", self._click)

    @property
    def value(self):
        return self._opts[self._index]

    def _draw(self):
        self.delete("all")
        p = self._px
        _round_rect(self, 0, 0, self._cw, self._ch, self._ch / 2,
                    fill=self._track, outline="")
        seg = (self._cw - p(8)) / len(self._opts)
        x = p(4) + seg * self._pos
        _round_rect(self, x, p(4), x + seg, self._ch - p(4),
                    (self._ch - p(8)) / 2, fill=self._pill, outline="")
        for i, opt in enumerate(self._opts):
            cx = p(4) + seg * i + seg / 2
            near = 1 - min(1.0, abs(self._pos - i))
            color = _mix(self._off_fg, self._on_fg, near)
            self.create_text(cx, self._ch / 2, text=opt, fill=color,
                             font=self._font_b if i == self._index else self._font)

    def _click(self, e):
        if not self._enabled:
            return
        seg = self._cw / len(self._opts)
        i = min(len(self._opts) - 1, max(0, int(e.x // seg)))
        if i != self._index:
            self._index = i
            self._slide(0.0, self._pos)
            self._cmd(self.value)

    def _slide(self, t, start):
        t = min(1.0, t + 0.09)
        self._pos = start + (self._index - start) * _ease(t)
        self._draw()
        if t < 1.0:
            self._job = self.after(14, self._slide, t, start)

    def set_enabled(self, on):
        self._enabled = on
        self.config(cursor="hand2" if on else "arrow")


class ActivityBar(tk.Canvas):

    def __init__(self, parent, height, bg, track, fill):
        super().__init__(parent, height=height, bg=bg,
                         highlightthickness=0, bd=0)
        self._track, self._fill, self._h = track, fill, height
        self._pos, self._running, self._job = 0.0, False, None
        self._done_color = None
        self.bind("<Configure>", lambda e: self._draw())

    def _draw(self):
        self.delete("all")
        w = self.winfo_width() or 1
        h = self._h
        _round_rect(self, 0, 0, w, h, h / 2, fill=self._track, outline="")
        if self._done_color:
            _round_rect(self, 0, 0, w, h, h / 2, fill=self._done_color,
                        outline="")
            return
        if not self._running:
            return
        seg = max(w * 0.3, 80)
        x = _ease(self._pos) * (w + seg) - seg
        x1, x2 = max(0, x), min(w, x + seg)
        if x2 > x1:
            _round_rect(self, x1, 0, x2, h, h / 2, fill=self._fill,
                        outline="")

    def start(self):
        self._done_color = None
        if not self._running:
            self._running, self._pos = True, 0.0
            self._tick()

    def _tick(self):
        self._pos = (self._pos + 0.011) % 1.0
        self._draw()
        self._job = self.after(16, self._tick)

    def stop(self, color=None):
        self._running = False
        if self._job is not None:
            self.after_cancel(self._job)
            self._job = None
        self._done_color = color
        self._draw()


class Spinner(tk.Canvas):

    def __init__(self, parent, size, bg, color):
        super().__init__(parent, width=size, height=size, bg=bg,
                         highlightthickness=0, bd=0)
        self._size, self._color = size, color
        self._angle, self._job = 0, None

    def start(self):
        if self._job is None:
            self._tick()

    def _tick(self):
        self.delete("all")
        s, w = self._size, max(2, self._size // 8)
        box = (w, w, s - w, s - w)
        self.create_oval(*box, outline=RED_TINT, width=w)
        self.create_arc(*box, start=self._angle, extent=100, style="arc",
                        outline=self._color, width=w)
        self._angle = (self._angle - 12) % 360
        self._job = self.after(24, self._tick)

    def stop(self):
        if self._job is not None:
            self.after_cancel(self._job)
            self._job = None
        self.delete("all")


class StatusDot(tk.Canvas):

    def __init__(self, parent, size, bg):
        super().__init__(parent, width=size, height=size, bg=bg,
                         highlightthickness=0, bd=0)
        self._s, self._bg = size, bg
        self._state, self._phase, self._job = "idle", 0.0, None
        self._draw()

    def set_bg(self, bg):
        self._bg = bg
        self.config(bg=bg)
        self._draw()

    def set_state(self, state):
        self._state = state
        if state == "running" and self._job is None:
            self._pulse()
        elif state != "running" and self._job is not None:
            self.after_cancel(self._job)
            self._job = None
        self._draw()

    def _pulse(self):
        self._phase = (self._phase + 0.06) % 1.0
        self._draw()
        self._job = self.after(30, self._pulse)

    def _draw(self):
        self.delete("all")
        s = self._s
        c = s / 2
        if self._state == "running":
            import math
            k = (math.sin(self._phase * 2 * math.pi) + 1) / 2
            r = s * (0.28 + 0.2 * k)
            self.create_oval(c - r, c - r, c + r, c + r,
                             fill=_mix(RED_SOFT, RED_TINT, k), outline="")
            r2 = s * 0.2
            self.create_oval(c - r2, c - r2, c + r2, c + r2,
                             fill=RED, outline="")
            return
        color = {"idle": LINE_STRONG, "ok": OK, "fail": RED}[self._state]
        r = s * 0.24
        self.create_oval(c - r, c - r, c + r, c + r, fill=color, outline="")

    def _set_badge(self, text, fg, bg):
        cv = self.badge
        cv.delete("all")
        w, h = int(cv["width"]), int(cv["height"])
        _round_rect(cv, 1, 1, w - 1, h - 1, h / 2, fill=bg, outline="")
        cv.create_text(w / 2, h / 2, text=text, fill=fg, font=self.f_label)

    def _pulse_badge(self, text, t=0.0):
        k = 1 - abs(1 - 2 * t)
        self._set_badge(text, RED, _mix(WHITE, "#f9c9d1", k))
        self._badge_job = self.after(40, self._pulse_badge, text,
                                     (t + 0.03) % 1.0)

    def _stop_badge(self):
        if self._badge_job is not None:
            self.after_cancel(self._badge_job)
            self._badge_job = None

    def _render_reports(self):
        for w in self.list_inner.winfo_children():
            w.destroy()
        self.rows, self.details = [], []
        reports = self._active_reports()
        mode = self.mode.get()
        tabs = sum(len(r["sheets"]) for r in reports)
        self.nametowidget
        self.list_cv.yview_moveto(0)
        head_text = (f"{mode} · {len(reports)} reports · {tabs} tabs"
                     if reports else f"{mode} · nothing configured")
        self.badge.master.itemconfig(self.head_sub, text=head_text)

        if not reports:
            box = tk.Frame(self.list_inner, bg=CARD)
            box.pack(fill="x", pady=self.px(60))
            tk.Label(box, text=f"No {mode.lower()} reports yet", bg=CARD,
                     fg=SUBTLE, font=self.f_body_b).pack()
            tk.Label(box, text=f"Add them to "
                     f"{'MONTHLY_REPORTS' if mode == 'Monthly' else 'REPORTS'}"
                     " in the config.", bg=CARD, fg=MUTED,
                     font=self.f_small).pack(pady=(self.px(4), 0))

        for i, r in enumerate(reports):
            self.details.append([])
            self.after(45 * i, self._add_row, i, r)

    def _add_row(self, i, r):
        p = self.px
        row = tk.Frame(self.list_inner, bg=CARD, cursor="hand2")
        row.pack(fill="x")
        if i:
            tk.Frame(row, bg=LINE, height=1).pack(fill="x", padx=p(12))
        content = tk.Frame(row, bg=CARD)
        content.pack(fill="x", padx=p(8), pady=p(7))

        dot = StatusDot(content, p(18), CARD)
        dot.pack(side="left", padx=(p(2), p(8)))
        texts = tk.Frame(content, bg=CARD)
        texts.pack(side="left", fill="x", expand=True)
        name = tk.Label(texts, text=r["name"], bg=CARD, fg=TEXT,
                        font=self.f_body, anchor="w")
        name.pack(fill="x")
        n = len(r["sheets"])
        default_sub = (" · ".join(s["old_sheet"] for s in r["sheets"])
                       + f"   ({n} tab{'s' if n > 1 else ''})")
        sub = tk.Label(texts, text=self._clip(default_sub), bg=CARD,
                       fg=MUTED, font=self.f_tiny, anchor="w")
        sub.pack(fill="x")
        arrow = tk.Label(content, text="›", bg=CARD, fg=CARD,
                         font=(self.f_body[0], 14))
        arrow.pack(side="right", padx=(0, p(6)))

        widgets = [row, content, texts, name, sub, arrow]
        info = dict(dot=dot, name=name, sub=sub, arrow=arrow,
                    widgets=widgets, report=r, default_sub=default_sub)
        self.rows.append(info)

        def fade(t=0.0):
            t = min(1.0, t + 0.12)
            name.config(fg=_mix(CARD, TEXT, _ease(t)))
            sub.config(fg=_mix(CARD, MUTED, _ease(t)))
            if t < 1.0:
                row.after(16, fade, t)
        fade()

        def hover(on):
            bg = ROW_HOVER if on else CARD
            for w in widgets:
                w.config(bg=bg)
            dot.set_bg(bg)
            arrow.config(fg=RED if on else bg)
        for w in widgets + [dot]:
            w.bind("<Enter>", lambda e: hover(True))
            w.bind("<Leave>", lambda e: hover(False))
            w.bind("<Button-1>", lambda e, k=i: self._show_details(k))

    def _clip(self, text, n=58):
        return text if len(text) <= n else text[:n - 1] + "…"

    def _set_row(self, i, state=None, sub=None, color=None):
        if not (0 <= i < len(self.rows)):
            return
        row = self.rows[i]
        if state:
            row["dot"].set_state(state)
        if sub is not None:
            row["sub"].config(text=self._clip(sub), fg=color or MUTED)

    def _show_details(self, i):
        if not (0 <= i < len(self.rows)):
            return
        from tkinter import messagebox
        name = self.rows[i]["report"]["name"]
        lines = [l.strip("\n") for l in self.details[i] if str(l).strip()]
        if not lines:
            messagebox.showinfo(name, "Not run yet.\n\nTabs: "
                                + ", ".join(s["old_sheet"] for s in
                                            self.rows[i]["report"]["sheets"]))
            return
        text = "\n".join(lines[-40:])
        failed = any("ERROR" in l or "QC FAILED" in l for l in lines)
        (messagebox.showerror if failed else messagebox.showinfo)(name, text)

    @staticmethod
    def _summarise(lines, kind, ok):
        lines = [str(l).strip() for l in lines if str(l).strip()]
        if not ok:
            for l in lines:
                if l.startswith("ERROR"):
                    return l.replace("ERROR:", "").strip()
            probs = [l[2:] for l in lines if l.startswith("- ")]
            if probs:
                return "; ".join(probs)
            return "Failed - click for details"
        if kind == "qc":
            cells = [l for l in lines if l.startswith("cells compared")]
            if cells:
                n = cells[-1].split(":")[1].split()[0]
                return f"QC passed · {n} cells match"
            return "QC passed"
        for l in lines:
            if "->" in l and ("renamed" in l or "saved" in l):
                return "Saved as " + l.split("->")[-1].strip()
            if l.startswith("updated:") or l.startswith("saved:"):
                return "Saved · " + l.split(":", 1)[1].strip()
        return "Done"

    def _drain_log(self):
        while True:
            try:
                kind, data = self.log_queue.get_nowait()
            except queue.Empty:
                break
            if kind == "done":
                self._finish(data)
            elif kind == "status":
                self.status.config(text=str(data), fg=RED)
            elif kind == "line":
                i, text = data
                if 0 <= i < len(self.details):
                    self.details[i].append(text)
            elif kind == "row":
                i, state, sub, color = data
                self._set_row(i, state, sub, color)
            elif kind == "email_done":
                self.email_btn.set_text("Email")
                self.email_btn.set_enabled(True)
            elif kind == "msg":
                self._show_toast(*data)
        self.after(80, self._drain_log)

    def _tick_elapsed(self):
        if not self.running:
            return
        mins, secs = divmod(int(time.time() - self._t0), 60)
        self.elapsed.config(text=f"{mins:d}:{secs:02d}")
        self._elapsed_job = self.after(500, self._tick_elapsed)

    def _show_toast(self, text, good):
        if self._toast is not None and self._toast.winfo_exists():
            self._toast.destroy()
        p = self.px
        color = OK if good else RED
        soft = OK_SOFT if good else RED_SOFT
        t = tk.Frame(self, bg=soft, highlightbackground=color,
                     highlightthickness=1)
        tk.Frame(t, bg=color, width=p(4)).pack(side="left", fill="y")
        tk.Label(t, text=("✓  " if good else "✕  ") + text, bg=soft,
                 fg=color, font=self.f_body_b, padx=p(14),
                 pady=p(9)).pack(side="left")
        self._toast = t
        start, target = p(self.H) + p(10), p(self.H) - p(128)

        def slide(y):
            t.place(relx=0.5, y=y, anchor="n")
            if y > target:
                t.after(12, slide, max(target, y - p(9)))
            else:
                t.after(3000, lambda: t.destroy() if t.winfo_exists() else None)
        slide(start)

    def _draft_email(self):
        self.email_btn.set_text("Opening…")
        self.email_btn.set_enabled(False)
        folder = self.folder.get()
        rdate = report_end_date(GENERATED)

        def work():
            try:
                make_draft_email(folder, report_date=rdate,
                                 to=_setting("EMAIL_TO", ""),
                                 cc=_setting("EMAIL_CC", ""),
                                 log=lambda m: None)
            except Exception as e:

    def _start(self):
        self._start_job("run")

    def _start_qc(self):
        self._start_job("qc")

    def _start_job(self, kind):
        if self.running or not self._active_reports():
            return
        folder = Path(self.folder.get())
        if not folder.is_dir():
            self._show_toast("Folder does not exist", False)
            return

        self.running = True
        self.job_kind = kind
        for i, row in enumerate(self.rows):
            row["dot"].set_state("idle")
            row["sub"].config(text=self._clip("Waiting…"), fg=MUTED)
            self.details[i] = []

        self.run_btn.set_text("Running…" if kind == "run" else "Run Automation")
        self.qc_btn.set_text("Checking…" if kind == "qc" else "Run QC")
        for b in (self.run_btn, self.qc_btn, self.email_btn):
            b.set_enabled(False)
        self.seg.set_enabled(False)
        self.browse_btn.config(fg=MUTED, cursor="arrow")

        self.status.config(text="Starting…", fg=RED)
        self._stop_badge()
        self._pulse_badge("RUNNING" if kind == "run" else "CHECKING")
        self.bar.start()
        self.spinner.start()
        self._t0 = time.time()
        self._tick_elapsed()

        threading.Thread(target=self._worker,
                         args=(folder, kind, self.mode.get().lower()),
                         daemon=True).start()

    def _worker(self, folder, kind="run", mode="weekly"):
        put = self.log_queue.put
        globals()["FOLDER"] = folder
        if kind == "run":
            GENERATED.clear()

        reports = (_setting("MONTHLY_REPORTS", [])
                   if mode == "monthly" else REPORTS)
        total = len(reports)
        ok = bad = 0
        for i, report in enumerate(reports):
            name = report.get("name", "?")
            verb = "Updating" if kind == "run" else "Checking"
            put(("status", f"{verb} {i + 1} of {total} · {name}"))
            put(("row", (i, "running",
                         "Updating…" if kind == "run" else "Checking…",
                         RED)))
            lines = []

            def log(m, i=i, lines=lines):
                lines.append(str(m))
                put(("line", (i, str(m))))

            try:
                if kind == "run":
                    process_report(report, log=log, mode=mode)
                    passed = True
                else:
                    passed = qc_report(report, log=log)
            except Exception as e:
                log(f"ERROR: {e}")
                passed = False

            if passed:
                ok += 1
            else:
                bad += 1
            put(("row", (i, "ok" if passed else "fail",
                         self._summarise(lines, kind, passed),
                         OK if passed else RED)))

        put(("done", (ok, bad)))

    def _finish(self, result):
        ok, bad = result
        kind = self.job_kind
        self.running = False
        self._stop_badge()
        self.spinner.stop()
        self.bar.stop(color=RED if bad else OK)
        if self._elapsed_job is not None:
            self.after_cancel(self._elapsed_job)
            self._elapsed_job = None

        self.run_btn.set_text("Run Automation")
        self.qc_btn.set_text("Run QC")
        self.run_btn.set_enabled(True)
        self.qc_btn.set_enabled(True)
        self.seg.set_enabled(True)
        self.browse_btn.config(fg=RED, cursor="hand2")
        self.email_btn.set_enabled(bool(GENERATED))

        if kind == "qc":
            summary = f"QC: {ok} passed, {bad} failed"
            badge = "QC FAILED" if bad else "QC PASSED"
        else:
            summary = f"{ok} updated, {bad} failed"
            badge = "FAILED" if bad else "DONE"

        hint = "  ·  click a red row for details" if bad else ""
        self.status.config(text=summary + hint, fg=RED if bad else OK)
        self._set_badge(badge, WHITE, RED_DEEP if bad else OK)
        self._show_toast(summary, good=not bad)


def _cli_reports():
    if "--monthly" in sys.argv:
        return _setting("MONTHLY_REPORTS", []), "monthly"
    return REPORTS, "weekly"


def run_cli():
    reports, mode = _cli_reports()
    GENERATED.clear()
    ok = bad = 0
    for report in reports:
        try:
            process_report(report, mode=mode)
            ok += 1
        except Exception as e:
            bad += 1
            print(f"    ERROR ({report.get('name', '?')}): {e}")
    print(f"\nFinished ({mode}). {ok} succeeded, {bad} failed.")
    input("Press Enter to close...")


def run_qc_cli():
    reports, mode = _cli_reports()
    ok = bad = 0
    for report in reports:
        try:
            if qc_report(report):
                ok += 1
            else:
                bad += 1
        except Exception as e:
            bad += 1
            print(f"    ERROR ({report.get('name', '?')}): {e}")
    print(f"\nQC finished ({mode}). {ok} passed, {bad} failed.")
    input("Press Enter to close...")


if __name__ == "__main__":
    if "--qc" in sys.argv:
        run_qc_cli()
    elif "--cli" in sys.argv:
        run_cli()
    else:
        scale = _enable_hidpi()
        ReportRunner(scale).mainloop()
