#
# Single-file protocol:
#   REQUEST  → one file with multiple labeled sections separated by tags
#   RESPONSE → one file with matching section tags and Copilot answers
#
# Request format:
#   [SECTION:chart_Sheet1_Current vs Previous]
#   <prompt text>
#   [SECTION:chart_Sheet1_Percentage Change]
#   <prompt text>
#   [SECTION:threshold_analysis]
#   <prompt text>
#   [SECTION:executive_summary]
#   <prompt text>
#
# Response format (same tags, Copilot fills the content):
#   [SECTION:chart_Sheet1_Current vs Previous]
#   <copilot response>
#   [SECTION:chart_Sheet1_Percentage Change]
#   <copilot response>
#   ...

SECTION_TAG = "[SECTION:"
SECTION_END = "]"


def _get_username():
    try:
        return os.getlogin()
    except OSError:
        return getpass.getuser()


def _make_timestamp():
    return datetime.now().strftime("%Y%m%d_%H%M%S")


def build_combined_request(sections: dict[str, str]) -> str:
    """Build a single request string with labeled sections."""
    parts = []
    for key, prompt in sections.items():
        parts.append(f"{SECTION_TAG}{key}{SECTION_END}")
        parts.append(prompt.strip())
        parts.append("")  # blank line separator
    return "\n".join(parts)


def parse_combined_response(text: str) -> dict[str, str]:
    """Parse a single response file into section_key -> response_text."""
    results = {}
    current_key = None
    current_lines = []
    for line in text.split("\n"):
        stripped = line.strip()
        if stripped.startswith(SECTION_TAG) and stripped.endswith(SECTION_END):
            # Save previous section
            if current_key is not None:
                results[current_key] = "\n".join(current_lines).strip()
            current_key = stripped[len(SECTION_TAG):-len(SECTION_END)]
            current_lines = []
        else:
            current_lines.append(line)
    # Save last section
    if current_key is not None:
        results[current_key] = "\n".join(current_lines).strip()
    return results


def send_combined_request(sections: dict[str, str]) -> str:
    """Write all prompts to a single request file. Returns timestamp."""
    os.makedirs(COPILOT_REQUEST_FOLDER, exist_ok=True)
    username = _get_username()
    ts = _make_timestamp()
    filename = f"request_{username}_{ts}.txt"
    filepath = os.path.join(COPILOT_REQUEST_FOLDER, filename)
    content = build_combined_request(sections)
    with open(filepath, "w", encoding="utf-8") as f:
        f.write(content)
    return ts


def wait_for_combined_response(timestamp: str, timeout: int = None) -> dict[str, str]:
    """Poll for a single response file and parse all sections."""
    if timeout is None:
        timeout = COPILOT_TIMEOUT
    username = _get_username()
    pattern = os.path.join(
        COPILOT_RESPONSE_FOLDER,
        f"response_{username}_{timestamp}*.txt",
    )
    start = time.time()
    while time.time() - start < timeout:
        matches = glob.glob(pattern)
        if matches:
            with open(matches[0], "r", encoding="utf-8") as f:
                text = f.read()
            return parse_combined_response(text)
        time.sleep(COPILOT_POLL_INTERVAL)
    return {}


def send_and_wait_combined(sections: dict[str, str], timeout: int = None) -> dict[str, str]:
    """Send all prompts in one file, wait for one response, return parsed sections."""
    ts = send_combined_request(sections)
    return wait_for_combined_response(ts, timeout)


def check_folders_exist() -> tuple[bool, str]:
    issues = []
    if not os.path.isdir(COPILOT_REQUEST_FOLDER):
        issues.append(f"Request folder not found:\n{COPILOT_REQUEST_FOLDER}")
    if not os.path.isdir(COPILOT_RESPONSE_FOLDER):
        issues.append(f"Response folder not found:\n{COPILOT_RESPONSE_FOLDER}")
    if issues:
        return False, "\n\n".join(issues)
    return True, "Copilot folders are ready."

# ── File Detection ───────────────────────────────────────────────

@dataclass
class ReportFile:
    path: str
    name: str
    base_name: str
    date_str: str
    report_date: datetime
    frequency: str  # "monthly" or "weekly"


def _parse_date_suffix(name_no_ext: str) -> tuple[str, str, datetime, str] | None:
    # Try weekly first (6 digits: MMDDYY)
    m = re.match(r"^(.+?)_(\d{6})$", name_no_ext)
    if m:
        base, ds = m.group(1), m.group(2)
        try:
            dt = datetime.strptime(ds, "%m%d%y")
            return base, ds, dt, "weekly"
        except ValueError:
            pass
    # Try monthly (4 digits: MMYY)
    m = re.match(r"^(.+?)_(\d{4})$", name_no_ext)
    if m:
        base, ds = m.group(1), m.group(2)
        try:
            dt = datetime.strptime(ds, "%m%y")
            return base, ds, dt, "monthly"
        except ValueError:
            pass
    return None


def scan_folder(folder_path: str) -> dict[str, list[ReportFile]]:
    groups: dict[str, list[ReportFile]] = {}
    if not os.path.isdir(folder_path):
        return groups
    for fname in os.listdir(folder_path):
        if not fname.lower().endswith((".xlsx", ".xls", ".xlsm")):
            continue
        name_no_ext = os.path.splitext(fname)[0]
        parsed = _parse_date_suffix(name_no_ext)
        if parsed is None:
            continue
        base, ds, dt, freq = parsed
        rf = ReportFile(
            path=os.path.join(folder_path, fname),
            name=fname, base_name=base,
            date_str=ds, report_date=dt, frequency=freq,
        )
        groups.setdefault(base, []).append(rf)
    for base in groups:
        groups[base].sort(key=lambda r: r.report_date, reverse=True)
    return groups


def get_compare_options(frequency: str) -> list[str]:
    if frequency == "monthly":
        return ["Last month", "Last 3 months", "Last 6 months", "Custom"]
    else:

# ── Data Reading ─────────────────────────────────────────────────

def read_report(filepath: str) -> dict[str, pd.DataFrame]:
    raw = pd.read_excel(filepath, sheet_name=None, engine="openpyxl")
    cleaned = {}
    for name, df in raw.items():
        df = df.dropna(how="all").dropna(axis=1, how="all")
        for col in df.columns:
            if df[col].dtype == object:
                converted = pd.to_numeric(df[col], errors="coerce")
                if converted.notna().sum() > 0.5 * df[col].notna().sum():
                    df[col] = converted
        num_cols = df.select_dtypes(include=[np.number]).columns
        if len(num_cols) > 0:
            df = df.dropna(subset=num_cols, how="all")
        if not df.empty:
            cleaned[name] = df.reset_index(drop=True)
    return cleaned


def get_numeric_columns(df: pd.DataFrame) -> list[str]:
    return df.select_dtypes(include=[np.number]).columns.tolist()


# ── ML Comparison ────────────────────────────────────────────────

@dataclass
class ComparisonResult:
    sheet_name: str
    current_file: str
    previous_file: str
    summary: dict = field(default_factory=dict)
    anomalies: list = field(default_factory=list)
    variance_data: pd.DataFrame = None
    charts: list = field(default_factory=list)


def compare_reports(current_path: str, previous_path: str) -> list[ComparisonResult]:
    current_sheets = read_report(current_path)
    previous_sheets = read_report(previous_path)
    results = []

    common_sheets = set(current_sheets.keys()) & set(previous_sheets.keys())
    for sheet in sorted(common_sheets):
        df_curr = current_sheets[sheet]
        df_prev = previous_sheets[sheet]

        num_cols = list(
            set(get_numeric_columns(df_curr)) & set(get_numeric_columns(df_prev))
        )
        if not num_cols:
            continue

        result = ComparisonResult(
            sheet_name=sheet,
            current_file=os.path.basename(current_path),
            previous_file=os.path.basename(previous_path),
        )

        curr_sums = df_curr[num_cols].sum()
        prev_sums = df_prev[num_cols].sum()
        variance = curr_sums - prev_sums
        pct_change = ((curr_sums - prev_sums) / prev_sums.replace(0, np.nan) * 100).fillna(0)

        result.variance_data = pd.DataFrame({
            "Column": num_cols,
            "Current": curr_sums.values,
            "Previous": prev_sums.values,
            "Variance": variance.values,
            "% Change": pct_change.values.round(2),
        })

        result.summary = {
            "total_columns": len(num_cols),
            "increased": int((variance > 0).sum()),
            "decreased": int((variance < 0).sum()),
            "unchanged": int((variance == 0).sum()),
            "max_increase_col": num_cols[variance.values.argmax()] if len(num_cols) > 0 else "N/A",
            "max_decrease_col": num_cols[variance.values.argmin()] if len(num_cols) > 0 else "N/A",
            "max_increase_pct": float(pct_change.max()),
            "max_decrease_pct": float(pct_change.min()),
        }

        if len(num_cols) >= 2 and len(df_curr) > 5:
            try:
                scaler = StandardScaler()
                curr_scaled = scaler.fit_transform(df_curr[num_cols].fillna(0))
                iso = IsolationForest(contamination=0.1, random_state=42)
                labels = iso.fit_predict(curr_scaled)
                anomaly_rows = df_curr[labels == -1]
                result.anomalies = anomaly_rows.head(10).to_dict("records")
            except Exception:
                result.anomalies = []

        results.append(result)
    return results


# ── Threshold Detection ──────────────────────────────────────────

def compute_threshold(files: list[str], sheet_name: str = None) -> dict[str, tuple[float, float, float]]:
    all_sums = {}
    for fpath in files:
        sheets = read_report(fpath)
        for sname, df in sheets.items():
            if sheet_name and sname != sheet_name:
                continue
            for col in get_numeric_columns(df):
                key = f"{sname}__{col}"
                all_sums.setdefault(key, []).append(df[col].sum())
    thresholds = {}
    for key, values in all_sums.items():
        arr = np.array(values)
        mean = arr.mean()
        std = arr.std()
        thresholds[key] = (mean, mean + 2 * std, mean - 2 * std)
    return thresholds


def check_threshold_breaches(current_path: str, threshold_files: list[str]) -> list[dict]:
    thresholds = compute_threshold(threshold_files)
    current_sheets = read_report(current_path)
    breaches = []
    for sname, df in current_sheets.items():
        for col in get_numeric_columns(df):
            key = f"{sname}__{col}"
            if key not in thresholds:
                continue
            mean, upper, lower = thresholds[key]
            current_val = df[col].sum()
            if current_val > upper:
                breaches.append({
                    "sheet": sname, "column": col,
                    "current": round(current_val, 2), "mean": round(mean, 2),
                    "upper_threshold": round(upper, 2), "direction": "above",
                    "deviation_pct": round((current_val - mean) / mean * 100 if mean != 0 else 0, 2),
                })
            elif current_val < lower:
                breaches.append({
                    "sheet": sname, "column": col,
                    "current": round(current_val, 2), "mean": round(mean, 2),
                    "lower_threshold": round(lower, 2), "direction": "below",
                    "deviation_pct": round((current_val - mean) / mean * 100 if mean != 0 else 0, 2),
                })
    return breaches


# ── Chart Generation ─────────────────────────────────────────────

def _fig_to_base64(fig) -> str:
    buf = io.BytesIO()
    fig.savefig(buf, format="png", dpi=120, bbox_inches="tight",
                facecolor="#1A1A2E", edgecolor="none")
    buf.seek(0)
    b64 = base64.b64encode(buf.read()).decode("utf-8")
    plt.close(fig)
    return b64


def generate_variance_chart(result: ComparisonResult) -> str:
    sns.set_theme(style="darkgrid")
    df = result.variance_data
    if df is None or df.empty:
        return ""
    df_sorted = df.reindex(df["Variance"].abs().sort_values(ascending=False).index)
    df_plot = df_sorted.head(10)

    fig, ax = plt.subplots(figsize=(10, 5))
    fig.patch.set_facecolor("#1A1A2E")
    ax.set_facecolor("#16213E")
    x = np.arange(len(df_plot))
    width = 0.35
    ax.bar(x - width / 2, df_plot["Previous"], width, label="Previous", color="#00B4D8", alpha=0.85)
    ax.bar(x + width / 2, df_plot["Current"], width, label="Current", color="#0033A0", alpha=0.85)
    ax.set_xticks(x)
    ax.set_xticklabels(df_plot["Column"], rotation=45, ha="right", color="#E8E8E8", fontsize=8)
    ax.set_ylabel("Value", color="#E8E8E8")
    ax.set_title(f"Current vs Previous - {result.sheet_name}", color="#E8E8E8", fontsize=12, fontweight="bold")
    ax.legend(facecolor="#16213E", edgecolor="#333", labelcolor="#E8E8E8")
    ax.tick_params(colors="#9E9E9E")
    for spine in ax.spines.values():
        spine.set_color("#333")
    return _fig_to_base64(fig)


def generate_pct_change_chart(result: ComparisonResult) -> str:
    sns.set_theme(style="darkgrid")
    df = result.variance_data
    if df is None or df.empty:
        return ""
    df_sorted = df.reindex(df["% Change"].abs().sort_values(ascending=False).index)
    df_plot = df_sorted.head(10)

    fig, ax = plt.subplots(figsize=(10, 5))
    fig.patch.set_facecolor("#1A1A2E")
    ax.set_facecolor("#16213E")
    colors = ["#C62828" if v < 0 else "#2E7D32" for v in df_plot["% Change"]]
    ax.barh(df_plot["Column"], df_plot["% Change"], color=colors, alpha=0.85)
    ax.set_xlabel("% Change", color="#E8E8E8")
    ax.set_title(f"Percentage Change - {result.sheet_name}", color="#E8E8E8", fontsize=12, fontweight="bold")
    ax.tick_params(colors="#9E9E9E")
    ax.invert_yaxis()
    for spine in ax.spines.values():
        spine.set_color("#333")
    return _fig_to_base64(fig)


def generate_anomaly_heatmap(result: ComparisonResult) -> str:
    sns.set_theme(style="darkgrid")
    df = result.variance_data
    if df is None or df.empty or len(df) < 2:
        return ""
    fig, ax = plt.subplots(figsize=(10, 4))
    fig.patch.set_facecolor("#1A1A2E")
    heat_data = df[["Column", "Variance", "% Change"]].set_index("Column").T
    sns.heatmap(
        heat_data.head(10) if heat_data.shape[1] > 10 else heat_data,
        annot=True, fmt=".1f", cmap="RdYlGn", center=0,
        ax=ax, cbar_kws={"label": "Value"}, linewidths=0.5, linecolor="#333",
    )
    ax.set_title(f"Variance Heatmap - {result.sheet_name}", color="#E8E8E8", fontsize=12, fontweight="bold")
    ax.tick_params(colors="#9E9E9E")
    return _fig_to_base64(fig)


# ── Copilot Prompt Builders ──────────────────────────────────────

def build_chart_description_prompt(result: ComparisonResult, chart_type: str) -> str:
    df = result.variance_data
    if df is None or df.empty:
        return ""
    top_changes = df.nlargest(3, "Variance", keep="first")
    bottom_changes = df.nsmallest(3, "Variance", keep="first")
    top_str = ", ".join(
        f"{r[0]} (+{r[1]:.1f}%)" for r in zip(top_changes["Column"], top_changes["% Change"])
    )
    bot_str = ", ".join(
        f"{r[0]} ({r[1]:.1f}%)" for r in zip(bottom_changes["Column"], bottom_changes["% Change"])
    )
    prompt = (
        f"You are a data analyst assistant. "
        f"Describe the following {chart_type} chart in 2-3 sentences for a stakeholder report.\n\n"
        f"Chart: Comparing current period vs previous period for sheet '{result.sheet_name}'.\n"
        f"Top increases: {top_str}.\n"
        f"Top decreases: {bot_str}.\n"
        f"Total columns compared: {len(df)}. "
        f"Increased: {result.summary.get('increased', 0)}, "
        f"Decreased: {result.summary.get('decreased', 0)}.\n\n"
        f"Write a concise, professional description."
    )
    return prompt


def build_threshold_breach_prompt(breaches: list[dict]) -> str:
    if not breaches:
        return ""
    lines = []
    for b in breaches[:5]:
        lines.append(
            f"- Column '{b['column']}' in sheet '{b['sheet']}' is "
            f"{b['deviation_pct']}% {b['direction']} the historical average "
            f"(current: {b['current']}, avg: {b['mean']})."
        )
    prompt = (
        "You are a business analyst. The following data columns have crossed "
        "their historical threshold (mean +/- 2 standard deviations) compared "
        "to the last 3 reporting periods:\n\n"
        + "\n".join(lines)
        + "\n\nFor each breach, suggest a possible business reason "
        "(market event, seasonality, data issue) in 1-2 sentences. "
        "Be specific about what could have caused the change."
    )
    return prompt


def build_summary_prompt(results: list[ComparisonResult], breaches: list[dict]) -> str:
    sections = []
    for r in results:
        s = r.summary
        sections.append(
            f"Sheet '{r.sheet_name}': {s.get('total_columns', 0)} columns compared. "
            f"{s.get('increased', 0)} increased, {s.get('decreased', 0)} decreased. "
            f"Largest increase: {s.get('max_increase_col', 'N/A')} ({s.get('max_increase_pct', 0):.1f}%). "
            f"Largest decrease: {s.get('max_decrease_col', 'N/A')} ({s.get('max_decrease_pct', 0):.1f}%)."
        )
    breach_lines = []
    for b in breaches[:5]:
        breach_lines.append(f"- {b['column']} ({b['sheet']}): {b['deviation_pct']}% {b['direction']} average")
    prompt = (
        "Write a professional stakeholder summary email (3-4 paragraphs) "
        "based on the following report comparison results:\n\n"
        "COMPARISON RESULTS:\n" + "\n".join(sections) + "\n\n"
    )
    if breach_lines:
        prompt += "THRESHOLD BREACHES:\n" + "\n".join(breach_lines) + "\n\n"
    prompt += (
        "Include: key findings, areas of concern, and recommended next steps. "
        "Use a professional tone suitable for senior management."
    )
    return prompt

    # ── Run Analysis ─────────────────────────────────────────────

    def _run_analysis(self, compare_option):
        files = self.report_groups.get(self.selected_group)
        if not files or len(files) < 2:
            messagebox.showwarning("Not enough files",
                                   "At least 2 report files are needed for comparison.")
            return

        current = files[0]
        if "last month" in compare_option.lower() or "last week" in compare_option.lower():
            previous = files[1] if len(files) > 1 else None
        elif "3" in compare_option:
            previous = files[2] if len(files) > 2 else files[-1]
        elif "6" in compare_option:
            previous = files[5] if len(files) > 5 else files[-1]
        else:
            previous = files[1]

        if previous is None:
            messagebox.showwarning("Not enough data", "Not enough files for this range.")
            return

        self._show_loading("Running ML analysis...")

        def worker():
            try:
                results = compare_reports(current.path, previous.path)
                self.analysis_results = results

                for r in results:
                    r.charts = []
                    c1 = generate_variance_chart(r)
                    if c1:
                        r.charts.append(("Current vs Previous", c1))
                    c2 = generate_pct_change_chart(r)
                    if c2:
                        r.charts.append(("Percentage Change", c2))
                    c3 = generate_anomaly_heatmap(r)
                    if c3:
                        r.charts.append(("Variance Heatmap", c3))

                threshold_files = [f.path for f in files[1:4]]
                if len(threshold_files) >= 1:
                    self.breaches = check_threshold_breaches(current.path, threshold_files)

                self._update_loading("Contacting Copilot agent...")
                ok, msg = check_folders_exist()
                if ok:
                    # Build all prompts into a single request
                    sections = {}
                    for r in results:
                        for chart_title, _ in r.charts:
                            prompt = build_chart_description_prompt(r, chart_title)
                            if prompt:
                                key = f"chart_{r.sheet_name}_{chart_title}"
                                sections[key] = prompt

                    if self.breaches:
                        breach_prompt = build_threshold_breach_prompt(self.breaches)
                        if breach_prompt:
                            sections["threshold_analysis"] = breach_prompt

                    summary_prompt = build_summary_prompt(results, self.breaches)
                    if summary_prompt:
                        sections["executive_summary"] = summary_prompt

                    # Send 1 file, wait for 1 response
                    self._update_loading("Waiting for Copilot response...")
                    responses = send_and_wait_combined(sections, timeout=120)

                    # Map responses back
                    for r in results:
                        for chart_title, _ in r.charts:
                            key = f"chart_{r.sheet_name}_{chart_title}"
                            if key in responses and responses[key]:
                                self.copilot_descriptions[f"{r.sheet_name}_{chart_title}"] = responses[key]

                    if "threshold_analysis" in responses and responses["threshold_analysis"]:
                        self.copilot_descriptions["threshold_analysis"] = responses["threshold_analysis"]

                    if "executive_summary" in responses and responses["executive_summary"]:
                        self.summary_text = responses["executive_summary"]
                    else:
                        self.summary_text = self._generate_local_summary(results, self.breaches)
                else:
                    self.summary_text = self._generate_local_summary(results, self.breaches)

                self.after(0, self._show_results)
            except Exception as e:
                self.after(0, lambda: self._show_error(str(e)))

        threading.Thread(target=worker, daemon=True).start()

    def _generate_local_summary(self, results, breaches):
        lines = [f"Report Comparison Summary - {APP_NAME}\n", "=" * 50]
        for r in results:
            s = r.summary
            lines.append(f"\nSheet: {r.sheet_name}")
            lines.append(f"  Compared: {r.current_file} vs {r.previous_file}")
            lines.append(f"  Columns: {s['total_columns']}  (+{s['increased']} / -{s['decreased']} / ={s['unchanged']})")
            lines.append(f"  Largest increase: {s['max_increase_col']} ({s['max_increase_pct']:.1f}%)")
            lines.append(f"  Largest decrease: {s['max_decrease_col']} ({s['max_decrease_pct']:.1f}%)")
        if breaches:
            lines.append(f"\n{'=' * 50}")
            lines.append("THRESHOLD BREACHES:")
            for b in breaches:
                lines.append(f"  - {b['column']} ({b['sheet']}): {b['deviation_pct']}% {b['direction']} average  (current={b['current']}, avg={b['mean']})")
        lines.append(f"\n{'=' * 50}")
        lines.append("Note: Copilot agent was not reachable. Connect OneDrive for AI-generated insights.")
        return "\n".join(lines)

    # ── Export to Outlook ────────────────────────────────────────

    def _export_to_outlook(self):
        subject = f"Insight IQ Analysis Report - {self.selected_group or 'Report'}"
        body = self.summary_text or "Analysis summary not available."

        if self.breaches:
            body += "\n\n--- THRESHOLD BREACHES ---\n"
            for b in self.breaches:
                body += f"- {b['column']} ({b['sheet']}): {b['deviation_pct']}% {b['direction']} average\n"

        try:
            import win32com.client
            outlook = win32com.client.Dispatch("Outlook.Application")
            mail = outlook.CreateItem(0)
            mail.Subject = subject
            mail.Body = body
            mail.Display()
        except ImportError:
            messagebox.showinfo("Export",
                                "Outlook integration requires Windows with pywin32.\n\n"
                                "Summary has been copied to clipboard instead.")
            self.clipboard_clear()
            self.clipboard_append(f"Subject: {subject}\n\n{body}")
        except Exception as e:
            messagebox.showerror("Outlook Error", str(e))

    # ── Browse Folder ────────────────────────────────────────────

    def _browse_folder(self):
        folder = filedialog.askdirectory(title="Select Project Folder")
        if not folder:
            return
        self.project_folder = folder
        display = os.path.basename(folder) or folder
        self.folder_label.configure(text=display, text_color=COLOR_PRIMARY)
        self.report_groups = scan_folder(folder)
        self._populate_report_selector()
        self._show_page("data_analysis")
