def col_number(ref):
    if isinstance(ref, int):
        return ref
    n = 0
    for ch in str(ref).strip().upper():
        if not ch.isalpha():
            raise ValueError(f"Bad column reference: {ref!r}")
        n = n * 26 + (ord(ch) - 64)
    return n


def col_letter(n):
    out = ""
    while n > 0:
        n, rem = divmod(n - 1, 26)
        out = chr(65 + rem) + out
    return out


def last_row(sheet):
    ur = sheet.api.UsedRange
    return ur.Row + ur.Rows.Count - 1


def last_col(sheet):
    ur = sheet.api.UsedRange
    return ur.Column + ur.Columns.Count - 1


def find_marker_row(sheet, start_row, marker):
    end_r, end_c = last_row(sheet), last_col(sheet)
    if end_r < start_row:
        return None

    needle = marker.strip()

    try:
        escaped = (needle.replace("~", "~~")
                         .replace("*", "~*")
                         .replace("?", "~?"))
        rng = sheet.api.Range(sheet.api.Cells(start_row, 1),
                              sheet.api.Cells(end_r, end_c))
        last_cell = rng.Cells(rng.Rows.Count, rng.Columns.Count)
        hit = rng.Find(escaped, last_cell, XL_VALUES, XL_PART,
                       XL_BY_ROWS, XL_NEXT, False)
        if hit is not None:
            return hit.Row
        return None
    except Exception:
        pass

    low = needle.lower()
    best = None
    for c in range(1, end_c + 1):
        vals = sheet.range((start_row, c), (end_r, c)).value
        if not isinstance(vals, list):
            vals = [vals]
        for i, v in enumerate(vals):
            if v is not None and low in str(v).lower():
                r = start_row + i
                if best is None or r < best:
                    best = r
                break
    return best


def find_disclaimer_top(sheet, marker_row, data_start, configured_above=0,
                        max_scan=15):
    end_c = last_col(sheet)
    r = marker_row
    for _ in range(max_scan):
        above = r - 1
        if above <= data_start:
            break
        vals = sheet.range((above, 1), (above, end_c)).value
        if not isinstance(vals, list):
            vals = [vals]
        if all(v is None or (isinstance(v, str) and not v.strip())
               for v in vals):
            return r
        r = above

    return marker_row - configured_above

def find_webi_last_row(src_ws, key_col, src_start):

def paste_values_in_chunks(source_ws, destination_ws,
                           source_start_row, source_end_row,
                           source_first_col, source_last_col,
                           destination_start_row, destination_first_col,
                           chunk_size=10000, log=print):
    total_rows = source_end_row - source_start_row + 1
    width = source_last_col - source_first_col + 1
    dst_col2 = destination_first_col + width - 1

    copied = 0
    while copied < total_rows:
        rows_this_chunk = min(chunk_size, total_rows - copied)

        src_row1 = source_start_row + copied
        src_row2 = src_row1 + rows_this_chunk - 1

        dst_row1 = destination_start_row + copied
        dst_row2 = dst_row1 + rows_this_chunk - 1

        source_range = source_ws.range((src_row1, source_first_col),
                                       (src_row2, source_last_col))
        destination_range = destination_ws.range((dst_row1, destination_first_col),
                                                 (dst_row2, dst_col2))

        if source_range.rows.count != destination_range.rows.count:
            raise ValueError(f"Row mismatch: source {src_row1}:{src_row2}, "
                             f"destination {dst_row1}:{dst_row2}")
        if source_range.columns.count != destination_range.columns.count:
            raise ValueError(f"Column mismatch in chunk starting at "
                             f"source row {src_row1}")

        destination_range.api.Value2 = source_range.api.Value2

        copied += rows_this_chunk
        log(f"    copied {copied:,} / {total_rows:,} rows")


def _as_number(v):
    if isinstance(v, bool):
        return None
    if isinstance(v, (int, float)):
        return float(v)
    if isinstance(v, str):
        t = v.strip().replace(",", "")
        try:
            return float(t)
        except ValueError:
            return None
    return None


def _qc_compare(a, b):
    blank_a = a is None or (isinstance(a, str) and a.strip() == "")
    blank_b = b is None or (isinstance(b, str) and b.strip() == "")
    if blank_a and blank_b:
        return "same"
    if blank_a != blank_b:
        return "diff"
    if a == b:
        return "same"

    na, nb = _as_number(a), _as_number(b)
    if na is not None and nb is not None:
        if abs(na - nb) <= 1e-9 * max(1.0, abs(na), abs(nb)):
            return "type" if type(a) is not type(b) else "same"
        return "diff"

    if isinstance(a, str) and isinstance(b, str) and a.strip() == b.strip():
        return "same"
    return "diff"


def _grid(values):
    if values is None or not isinstance(values, (list, tuple)):
        return [[values]]
    if values and not isinstance(values[0], (list, tuple)):
        return [list(values)]
    return [list(r) for r in values]


def qc_sheet(ws, src_ws, start, c1, c2, src_start, src_last_row, src_c1,
             src_c2, disclaimer_text, block_offset, blank_rows, log=print,
             chunk_size=10000, max_report=10):
    name = ws.name
    rows = src_last_row - src_start + 1
    last = start + rows - 1
    problems = []

    diffs, type_diffs, shown = 0, 0, 0
    done = 0
    while done < rows:
        n = min(chunk_size, rows - done)
        a = _grid(ws.range((start + done, c1),
                           (start + done + n - 1, c2)).api.Value2)
        b = _grid(src_ws.range((src_start + done, src_c1),
                               (src_start + done + n - 1, src_c2)).api.Value2)

        for i, (ra, rb) in enumerate(zip(a, b)):
            for j, (va, vb) in enumerate(zip(ra, rb)):
                res = _qc_compare(va, vb)
                if res == "same":
                    continue
                if res == "type":
                    type_diffs += 1
                    continue
                diffs += 1
                if shown < max_report:
                    shown += 1
                    log(f"      diff {col_letter(c1 + j)}{start + done + i}: "
                        f"report {va!r} | webi {vb!r} "
                        f"(webi {col_letter(src_c1 + j)}"
                        f"{src_start + done + i})")
        done += n

    if diffs:
        problems.append(f"{diffs:,} cell(s) differ")
    if type_diffs:
        log(f"    QC note: {type_diffs:,} cell(s) match in value but one "
            "side is stored as text - check the column number format.")

    below = _grid(ws.range((last + 1, c1), (last + 1, c2)).api.Value2)[0]
    if any(v not in (None, "") for v in below):
        problems.append(f"row {last + 1} (should be blank) has data: {below}")

    marker = find_marker_row(ws, last + 1, disclaimer_text)
    if marker is None:
        problems.append("disclaimer not found below the data")
    else:
        top = marker - block_offset
        expected = last + blank_rows + 1
        if top != expected:
            problems.append(f"disclaimer starts at row {top}, "
                            f"expected row {expected}")

    for label, r_rep, r_src in (("first", start, src_start),
                                ("last ", last, src_last_row)):
        rep = _grid(ws.range((r_rep, c1), (r_rep, c2)).api.Value2)[0]
        src = _grid(src_ws.range((r_src, src_c1), (r_src, src_c2)).api.Value2)[0]
        log(f"    QC {label} row  report {r_rep}: {rep}")
        log(f"    QC {label} row  webi   {r_src}: {src}")

    cells = rows * (c2 - c1 + 1)
    if problems:
        log(f"    QC FAILED on '{name}': " + "; ".join(problems))
        return False

    log(f"    QC PASSED on '{name}': {rows:,} rows, {cells:,} cells match, "
        f"disclaimer {blank_rows} row(s) below data")
    return True


def webi_tab(webi_wb, cfg):
    """WebI tab for a config. A CSV WebI has one tab, so webi_sheet is
    optional there (Excel names that tab after the file)."""
    names = [x.name for x in webi_wb.sheets]
    want = cfg.get("webi_sheet")
    if want and want in names:
        return webi_wb.sheets[want]
    is_csv = str(getattr(webi_wb, "name", "")).lower().endswith(".csv")
    if (is_csv or not want) and len(names) == 1:
        return webi_wb.sheets[0]
    raise ValueError(f"WebI tab '{want}' not found. Tabs: {names}")


def process_sheet(wb, webi_wb, cfg, log=print):
    name = cfg["old_sheet"]
    blank_rows = BLANK_ROWS_BEFORE_DISCLAIMER
    above_marker = cfg.get("disclaimer_rows_above_marker", 0) or 0

    sheet_names = [s.name for s in wb.sheets]
    if name not in sheet_names:
        raise ValueError(f"Tab '{name}' not found. Tabs: {sheet_names}")

    ws = wb.sheets[name]
    start = cfg["old_data_start_row"]
    c1 = col_number(cfg["old_first_col"])
    c2 = col_number(cfg["old_last_col"])
    width = c2 - c1 + 1

    src_ws = webi_tab(webi_wb, cfg)
    webi_sheet_name = src_ws.name
    log(f"  Tab '{name}'  <-  WebI '{webi_sheet_name}'")
    src_start = cfg["webi_data_start_row"]
    src_c1 = col_number(cfg["webi_first_col"])
    src_c2 = col_number(cfg["webi_last_col"])
    src_width = src_c2 - src_c1 + 1

    if src_width != width:
        raise ValueError(
            f"Column count mismatch on '{name}': WebI gives {src_width} "
            f"({cfg['webi_first_col']}-{cfg['webi_last_col']}) but report "
            f"expects {width} ({cfg['old_first_col']}-{cfg['old_last_col']})."
        )

    key_col = col_number(cfg.get("webi_key_col", cfg["webi_first_col"]))
    src_last_row = find_webi_last_row(src_ws, key_col, src_start)

    if src_last_row < src_start:
        raise ValueError(f"No data found in WebI tab '{webi_sheet_name}' - "
                         "stopping so the old data isn't wiped.")

    new_count = src_last_row - src_start + 1

    check_start = max(src_start, src_last_row - 2)
    log(f"    WebI {col_letter(src_c1)}{src_start}:"
        f"{col_letter(src_c2)}{src_last_row}  ({new_count:,} rows)")
    try:
        tail = src_ws.range((check_start, src_c1),
                            (src_last_row, src_c2)).options(ndim=2).value
        for i, row in enumerate(tail, start=check_start):
            log(f"      WebI row {i}: {row}")
    except Exception:
        pass

    marker = find_marker_row(ws, start, cfg["disclaimer_marker_text"])
    if marker is None:
        raise ValueError(f"Disclaimer not found on '{name}'. "
                         "Check disclaimer_marker_text.")

    disc_start = find_disclaimer_top(ws, marker, start, above_marker)
    block_offset = marker - disc_start
    if disc_start <= start:
        raise ValueError(f"Disclaimer row {disc_start} is at/above data "
                         f"start row {start} on '{name}'.")

    old_count = max(0, disc_start - blank_rows - start)

    new_last_data = start + new_count - 1
    required_disc_start = new_last_data + blank_rows + 1

    log(f"    old rows: {old_count:,}   new rows: {new_count:,}")
    log(f"    disclaimer: row {disc_start:,} -> row {required_disc_start:,}")

    row_difference = required_disc_start - disc_start

    if row_difference > 0:
        ws.api.Rows(f"{disc_start}:{disc_start + row_difference - 1}").Insert()
        log(f"    inserted {row_difference:,} row(s)")
    elif row_difference < 0:
        n = -row_difference
        ws.api.Rows(f"{disc_start - n}:{disc_start - 1}").Delete()
        log(f"    deleted {n:,} row(s)")

    ws.range((start, c1), (required_disc_start - 1, c2)).clear_contents()

    paste_values_in_chunks(
        source_ws=src_ws,
        destination_ws=ws,
        source_start_row=src_start,
        source_end_row=src_last_row,
        source_first_col=src_c1,
        source_last_col=src_c2,
        destination_start_row=start,
        destination_first_col=c1,
        chunk_size=_setting("CHUNK_SIZE", 10000),
        log=log,
    )

    marker_after = find_marker_row(ws, new_last_data + 1,
                                   cfg["disclaimer_marker_text"])
    if marker_after is None:
        raise ValueError(f"Disclaimer lost after paste on '{name}'.")

    disc_top = marker_after - block_offset

    if disc_top - 1 >= new_last_data + 1:
        ws.range((new_last_data + 1, c1), (disc_top - 1, c2)).clear_contents()

    gap = disc_top - new_last_data - 1
    if gap < blank_rows:
        need = blank_rows - gap
        ws.api.Rows(f"{disc_top}:{disc_top + need - 1}").Insert()
        log(f"    fixed gap: inserted {need} row(s)")
    elif gap > blank_rows:
        extra = gap - blank_rows
        ws.api.Rows(f"{new_last_data + 1}:{new_last_data + extra}").Delete()
        log(f"    fixed gap: deleted {extra} row(s)")

    log(f"    final: data {col_letter(c1)}{start}:{col_letter(c2)}"
        f"{new_last_data}, disclaimer at row "
        f"{new_last_data + blank_rows + 1:,}")

    if _setting("QC_ENABLED", True):
        passed = qc_sheet(
            ws, src_ws,
            start=start, c1=c1, c2=c2,
            src_start=src_start, src_last_row=src_last_row,
            src_c1=src_c1, src_c2=src_c2,
            disclaimer_text=cfg["disclaimer_marker_text"],
            block_offset=block_offset,
            blank_rows=blank_rows,
            log=log,
        )
        if not passed and _setting("QC_BLOCKS_SAVE", True):
            raise ValueError(f"QC failed on '{name}' - report NOT saved. "
                             "See the diff lines above.")

    go_to_a1(ws)


def _row_is_blank(vals):
    return all(v is None or (isinstance(v, str) and not v.strip())
               for v in vals)


def _report_last_data_row(ws, start, c1, c2, disc_top):
    r = disc_top - 1
    while r >= start:
        if not _row_is_blank(_grid(ws.range((r, c1), (r, c2)).api.Value2)[0]):
            return r
        r -= 1
    return start - 1


def qc_tab_readonly(wb, webi_wb, cfg, log=print, max_report=10):
    name = cfg["old_sheet"]
    blank_rows = BLANK_ROWS_BEFORE_DISCLAIMER
    problems = []

    log(f"  QC tab '{name}'  vs  WebI '{cfg.get('webi_sheet', '')}'")

    if name not in [x.name for x in wb.sheets]:
        log(f"    tab '{name}' not found in report")
        log("    QC FAILED")
        return False
    try:
        src = webi_tab(webi_wb, cfg)
    except ValueError as e:
        log(f"    {e}")
        log("    QC FAILED")
        return False

    ws = wb.sheets[name]

    start = cfg["old_data_start_row"]
    c1 = col_number(cfg["old_first_col"])
    c2 = col_number(cfg["old_last_col"])
    s_start = cfg["webi_data_start_row"]
    s_c1 = col_number(cfg["webi_first_col"])
    s_c2 = col_number(cfg["webi_last_col"])

    if (c2 - c1) != (s_c2 - s_c1):
        log(f"    column count differs - report {c2 - c1 + 1}, "
            f"WebI {s_c2 - s_c1 + 1}")
        log("    QC FAILED")
        return False
    width = c2 - c1 + 1

    key_col = col_number(cfg.get("webi_key_col", cfg["webi_first_col"]))
    s_last = find_webi_last_row(src, key_col, s_start)
    webi_rows = max(0, s_last - s_start + 1)

    above = cfg.get("disclaimer_rows_above_marker", 0) or 0
    marker = find_marker_row(ws, start, cfg["disclaimer_marker_text"])
    if marker is None:
        problems.append("disclaimer not found")
        r_last = ws.api.Cells(ws.api.Rows.Count, c1).End(XL_UP).Row
    else:
        disc_top = find_disclaimer_top(ws, marker, start, above)
        r_last = _report_last_data_row(ws, start, c1, c2, disc_top)
        gap = disc_top - r_last - 1
        if gap != blank_rows:
            problems.append(f"gap before disclaimer is {gap} row(s), "
                            f"expected {blank_rows}")
    report_rows = max(0, r_last - start + 1)

    log(f"    rows: report {report_rows:,} | WebI {webi_rows:,}")
    if report_rows != webi_rows:
        problems.append(f"row count differs by {report_rows - webi_rows:+,}")

    n = min(report_rows, webi_rows)
    chunk = _setting("CHUNK_SIZE", 10000)
    diffs = type_diffs = shown = 0

    for off in range(0, n, chunk):
        k = min(chunk, n - off)
        a = _grid(ws.range((start + off, c1),
                           (start + off + k - 1, c2)).api.Value2)
        b = _grid(src.range((s_start + off, s_c1),
                            (s_start + off + k - 1, s_c2)).api.Value2)
        for i in range(k):
            for j in range(width):
                res = _qc_compare(a[i][j], b[i][j])
                if res == "same":
                    continue
                if res == "type":
                    type_diffs += 1
                    continue
                diffs += 1
                if shown < max_report:
                    shown += 1
                    log(f"      diff {col_letter(c1 + j)}{start + off + i}: "
                        f"report {a[i][j]!r} | WebI "
                        f"{col_letter(s_c1 + j)}{s_start + off + i} "
                        f"{b[i][j]!r}")

    log(f"    cells compared: {n * width:,}   differences: {diffs:,}")
    if diffs:
        problems.append(f"{diffs:,} cell(s) differ")
    if type_diffs:
        log(f"    note: {type_diffs:,} cell(s) equal in value but one side "
            "is stored as text")

    if n:
        for label, r_rep, r_src in (("first", start, s_start),
                                    ("last ", start + n - 1, s_start + n - 1)):
            rep = _grid(ws.range((r_rep, c1), (r_rep, c2)).api.Value2)[0]
            web = _grid(src.range((r_src, s_c1), (r_src, s_c2)).api.Value2)[0]
            log(f"    {label} row  report {r_rep}: {list(rep)}")
            log(f"    {label} row  webi   {r_src}: {list(web)}")

    if problems:
        for p_ in problems:
            log(f"    - {p_}")
        log("    QC FAILED")
        return False

    log("    QC PASSED")
    return True


def find_csv(prefix, folder):
    key = prefix.strip().lower()
    hits = [f for f in Path(folder).iterdir()
            if f.is_file() and f.suffix.lower() == ".csv"
            and not f.name.lower().startswith(WEBI_PREFIX.lower())
            and f.stem.strip().lower().startswith(key)]
    if len(hits) > 1:
        raise ValueError(f"Multiple CSV files start with '{prefix}': "
                         f"{[f.name for f in hits]}")
    return hits[0] if hits else None


def _csv_cell(v):
    if v is None:
        return ""
    if isinstance(v, datetime):
        return v.strftime(_setting("CSV_DATE_FORMAT", "%m/%d/%Y"))
    if isinstance(v, float) and v.is_integer():
        return str(int(v))
    return str(v)


def read_webi_rows(ws, start, c1, c2, last):
    rows = []
    chunk = _setting("CHUNK_SIZE", 10000)
    r = start
    while r <= last:
        r2 = min(last, r + chunk - 1)
        rows.extend(ws.range((r, c1), (r2, c2)).options(ndim=2).value)
        r = r2 + 1
    return rows


def _csv_source(webi_wb, cfg):
    ws = webi_tab(webi_wb, cfg)
    sheet = ws.name
    c1 = col_number(cfg["webi_first_col"])
    c2 = col_number(cfg["webi_last_col"])
    start = cfg["webi_data_start_row"]
    key = col_number(cfg.get("webi_key_col", cfg["webi_first_col"]))
    last = find_webi_last_row(ws, key, start)
    if last < start:
        raise ValueError(f"No data in WebI tab '{sheet}' for the CSV.")
    rows = read_webi_rows(ws, start, c1, c2, last)
    header = None
    hr = cfg.get("webi_header_row")
    if hr:
        header = ws.range((hr, c1), (hr, c2)).options(ndim=2).value[0]
    return header, rows


def write_csv_from_webi(webi_wb, folder, cfg, log=print):
    header, rows = _csv_source(webi_wb, cfg)
    target = find_csv(cfg["csv_file"], folder)
    if target is None:
        target = Path(folder) / f"{cfg['csv_file']}.csv"
    with open(target, "w", newline="",
              encoding=_setting("CSV_ENCODING", "utf-8-sig")) as fh:
        w = csv.writer(fh)
        if header:
            w.writerow([_csv_cell(v) for v in header])
        for row in rows:
            w.writerow([_csv_cell(v) for v in row])
    log(f"    csv: {target.name}  ({len(rows):,} rows from WebI "
        f"'{cfg.get('webi_sheet', 'csv')}')")
    return target


def qc_csv(webi_wb, folder, cfg, log=print, max_report=10):
    log(f"  QC csv '{cfg['csv_file']}'  vs  WebI "
        f"'{cfg.get('webi_sheet', 'csv')}'")
    target = find_csv(cfg["csv_file"], folder)
    if target is None:
        log("    no csv in folder - skipped")
        return True
    header, rows = _csv_source(webi_wb, cfg)
    expected = [[_csv_cell(v) for v in r] for r in rows]
    with open(target, newline="",
              encoding=_setting("CSV_ENCODING", "utf-8-sig")) as fh:
        got = list(csv.reader(fh))
    if header:
        got = got[1:]
    log(f"    rows: csv {len(got):,} | WebI {len(expected):,}")
    diffs = shown = 0
    for i, (a, b) in enumerate(zip(got, expected)):
        for j in range(max(len(a), len(b))):
            va = a[j] if j < len(a) else ""
            vb = b[j] if j < len(b) else ""
            if va.strip() != vb.strip():
                diffs += 1
                if shown < max_report:
                    shown += 1
                    log(f"      diff row {i + 1} col {j + 1}: csv {va!r} | "
                        f"WebI {vb!r}")
    problems = []
    if len(got) != len(expected):
        problems.append(f"row count differs by {len(got) - len(expected):+,}")
    if diffs:
        problems.append(f"{diffs:,} cell(s) differ")
    log(f"    cells compared: {sum(len(r) for r in expected):,}   "
        f"differences: {diffs:,}")
    if problems:
        for p_ in problems:
            log(f"    - {p_}")
        log("    QC FAILED")
        return False
    log("    QC PASSED")
    return True


def add_months(d, n):
    m = d.month - 1 + n
    return datetime(d.year + m // 12, m % 12 + 1, 1)


def month_fields(d):
    return {
        "year": d.year,
        "yy": f"{d.year % 100:02d}",
        "m": d.month,
        "mm": f"{d.month:02d}",
        "month": MONTH_NAMES[d.month - 1],
        "mon": MONTH_ABBR[d.month - 1],
    }


def month_title(d):
    return f"{MONTH_NAMES[d.month - 1]} {d.year}"


def _format_regex(fmt):
    rx = ""
    for part in re.split(r"(\{\w+\})", fmt):
        if part == "{year}":
            rx += r"(?P<year>\d{4})"
        elif part == "{yy}":
            rx += r"(?P<yy>\d{2})"
        elif part == "{month}":
            rx += "(?P<month>" + "|".join(MONTH_NAMES) + ")"
        elif part == "{mon}":
            rx += "(?P<mon>" + "|".join(MONTH_ABBR) + ")"
        elif part == "{mm}":
            rx += r"(?P<mm>\d{2})"
        elif part == "{m}":
            rx += r"(?P<m>\d{1,2})"
        else:
            rx += re.escape(part)
    return re.compile("^" + rx + "$", re.I)


def parse_month_folder(name):
    m = _format_regex(LOCAL_MONTH_FORMAT).match(name.strip())
    if not m:
        return None
    g = m.groupdict()
    year = int(g["year"]) if g.get("year") else 2000 + int(g["yy"])
    if g.get("month"):
        month = [x.lower() for x in MONTH_NAMES].index(g["month"].lower()) + 1
    elif g.get("mon"):
        month = [x.lower() for x in MONTH_ABBR].index(g["mon"].lower()) + 1
    else:
        month = int(g.get("mm") or g.get("m"))
    if not 1 <= month <= 12:
        return None
    return datetime(year, month, 1)

def _is_webi(f):
    return f.name.lower().startswith(WEBI_PREFIX.lower())


def _is_temp(f):
    return f.name.startswith("~$")


def scan_downloads(reports):
    folder = Path(DOWNLOADS)
    found = {}
    if not folder.is_dir():
        return found
    prefixes = [r["webi_file"].strip().lower() for r in reports]
    for f in folder.iterdir():
        if not f.is_file() or _is_temp(f):
            continue
        if f.suffix.lower() not in EXCEL_EXT + (".csv",):
            continue
        stem = f.stem.strip().lower()
        best = None
        for i, p in enumerate(prefixes):
            if stem.startswith(p) and (best is None or len(p) > len(prefixes[best])):
                best = i
        if best is None:
            continue
        prev = found.get(best)
        if prev is None or f.stat().st_mtime > prev.stat().st_mtime:
            found[best] = f
    return found


def create_month_folder(source, target, log=print):
    if target.exists():
        raise FileExistsError(f"{target.name} already exists - not copying.")
    log(f"    copying {source.name} -> {target.name}")
    shutil.copytree(source, target)
    n = sum(1 for _ in target.rglob("*") if _.is_file())
    log(f"    created {target.name} ({n:,} files)")


def clear_webi(folder, log=print):
    removed = 0
    for f in sorted(Path(folder).rglob("*")):
        if f.is_file() and _is_webi(f):
            f.unlink()
            removed += 1
            log(f"    deleted {f.relative_to(folder)}")
    log(f"    removed {removed} old WebI file(s)")
    return removed


def _top_files(folder):
    return [f for f in Path(folder).iterdir() if f.is_file() and not _is_temp(f)]


def find_webi_in(report, folder):
    key = report["webi_file"].strip().lower()
    hits = [f for f in _top_files(folder)
            if f.stem.strip().lower().startswith(key)]
    if len(hits) > 1:
        raise ValueError(f"Multiple WebI files in {folder.name}: "
                         f"{[f.name for f in hits]}")
    return hits[0] if hits else None


def find_report_file(report, folder):
    key = report.get("report_file", report["name"]).strip().lower()
    hits = [f for f in _top_files(folder)
            if f.suffix.lower() in EXCEL_EXT and not _is_webi(f)
            and f.stem.strip().lower().startswith(key)]
    if not hits:
        raise FileNotFoundError(f"No report starting with "
                                f"'{report.get('report_file', report['name'])}'"
                                f" in {folder}")
    if len(hits) > 1:
        raise ValueError(f"Multiple reports match in {folder.name}: "
                         f"{[f.name for f in hits]}")
    return hits[0]


def place_webi(report, folder, downloaded, log=print):
    if not folder.is_dir():
        raise FileNotFoundError(f"Report folder not found: {folder}")
    if downloaded is not None:
        old = find_webi_in(report, folder)
        if old is not None:
            old.unlink()
            log(f"    replaced old {old.name}")
        dest = folder / downloaded.name
        shutil.move(str(downloaded), str(dest))
        log(f"    moved {downloaded.name} from Downloads")
        return dest
    existing = find_webi_in(report, folder)
    if existing is None:
        raise FileNotFoundError(f"WebI '{report['webi_file']}' not in "
                                "Downloads or the report folder")
    log(f"    using {existing.name} already in folder")
    return existing

_NAME_TOKENS = sorted(MONTH_NAMES + MONTH_ABBR + ["Sept"], key=len,
                      reverse=True)
_MON_RX = re.compile(r"(?<![A-Za-z])(" + "|".join(_NAME_TOKENS)
                     + r")([ _\-]?)((?:19|20)\d{2})(?!\d)", re.I)
_NUM_RX = re.compile(r"(?<!\d)(0[1-9]|1[0-2])([ _\-]?)((?:19|20)\d{2})(?!\d)")
_YM_RX = re.compile(r"(?<!\d)((?:19|20)\d{2})([_\-])(0[1-9]|1[0-2])(?!\d)")


def _match_case(word, sample):
    if sample.isupper():
        return word.upper()
    if sample.islower():
        return word.lower()
    return word


def roll_name(stem, d):
    def by_name(m):
        tok = m.group(1)
        if tok.lower() in [x.lower() for x in MONTH_NAMES]:
            word = MONTH_NAMES[d.month - 1]
        elif tok.lower() == "sept":
            word = "Sept" if d.month == 9 else MONTH_ABBR[d.month - 1]
        else:
            word = MONTH_ABBR[d.month - 1]
        return _match_case(word, tok) + m.group(2) + str(d.year)

    out = _MON_RX.sub(by_name, stem)
    out = _NUM_RX.sub(lambda m: f"{d.month:02d}{m.group(2)}{d.year}", out)
    out = _YM_RX.sub(lambda m: f"{d.year}{m.group(2)}{d.month:02d}", out)
    return out


def rename_files(report, folder, target, log=print):
    d = add_months(target, report.get("name_month_offset", 0) or 0)
    renamed = 0
    for f in _top_files(folder):
        if _is_webi(f):
            continue
        new_stem = roll_name(f.stem, d)
        if new_stem == f.stem:
            continue
        dest = f.with_name(new_stem + f.suffix)
        if dest.exists():
            log(f"    skip rename, {dest.name} already exists")
            continue
        for _ in range(10):
            try:
                f.rename(dest)
                break
            except PermissionError:
                time.sleep(0.5)
        else:
            raise PermissionError(f"Could not rename {f.name} - is it open?")
        renamed += 1
        log(f"    renamed {f.name} -> {dest.name}")
    if not renamed:
        log("    no dates in file names to change")
    return renamed

def file_url(path):
    p = str(path)
    if p.startswith("\\\\"):
        url = "file://" + p[2:].replace("\\", "/")
    else:
        url = "file:///" + p.replace("\\", "/")
    return url


def email_groups(reports):
    groups = {}
    for i, r in enumerate(reports):
        g = r.get("email_group") or (1 if i < 5 else 2)
        groups.setdefault(g, []).append(r)
    return dict(sorted(groups.items()))


def draft_emails(reports, d, log=print):
    fields = month_fields(d)
    opened = 0
    for g, members in email_groups(reports).items():
        cfg = EMAIL_GROUPS.get(g, {})
        subject = cfg.get("subject", "Monthly Reports - {month} {year}")
        subject = subject.format(**fields)
        items = [(r["name"], file_url(network_dest(r, d))) for r in members]

        try:
            import win32com.client
            outlook = win32com.client.Dispatch("Outlook.Application")
            mail = outlook.CreateItem(0)
            mail.To = cfg.get("to", "")
            if cfg.get("cc"):
                mail.CC = cfg["cc"]
            mail.Subject = subject
            links = "<br>".join(
                f"{html.escape(name)}: <a href=\"{urllib.parse.quote(url, safe=':/')}\">"
                f"{html.escape(url)}</a>" for name, url in items)
            text = EMAIL_BODY.format(links="\x00LINKS\x00", **fields)
            body = html.escape(text).replace("\n", "<br>")
            mail.HTMLBody = ('<div style="font-family:Calibri,sans-serif;'
                             'font-size:11pt">'
                             + body.replace("\x00LINKS\x00", links) + "</div>")
            mail.Display(False)
            log(f"    email {g}: opened in Outlook ({len(items)} reports)")
        except Exception as e:
            plain = EMAIL_BODY.format(
                links="\n".join(f"{n}: {u}" for n, u in items), **fields)
            params = {"subject": subject, "body": plain}
            if cfg.get("cc"):
                params["cc"] = cfg["cc"]
            url = ("mailto:" + urllib.parse.quote(cfg.get("to", "")) + "?"
                   + urllib.parse.urlencode(params,
                                            quote_via=urllib.parse.quote))
            webbrowser.open(url)
            log(f"    email {g}: Outlook unavailable ({type(e).__name__}), "
                "opened default mail app")
        opened += 1
    return opened


def _enable_hidpi():
    if not sys.platform.startswith("win"):
        return 1.0
    try:
        import ctypes
        try:
            ctypes.windll.shcore.SetProcessDpiAwareness(1)
        except Exception:
            ctypes.windll.user32.SetProcessDPIAware()
        dc = ctypes.windll.user32.GetDC(0)
        dpi = ctypes.windll.gdi32.GetDeviceCaps(dc, 88)
        ctypes.windll.user32.ReleaseDC(0, dc)
        return (dpi or 96) / 96.0
    except Exception:
        return 1.0


def _mix(c1, c2, t):
    a = [int(c1[i:i + 2], 16) for i in (1, 3, 5)]
    b = [int(c2[i:i + 2], 16) for i in (1, 3, 5)]
    return "#%02x%02x%02x" % tuple(round(x + (y - x) * t) for x, y in zip(a, b))


def _ease(t):
    return t * t * (3 - 2 * t)


def _round_rect(cv, x1, y1, x2, y2, r, **kw):
    r = max(0, min(r, (x2 - x1) / 2, (y2 - y1) / 2))
    pts = [x1 + r, y1, x2 - r, y1, x2, y1, x2, y1 + r,
           x2, y2 - r, x2, y2, x2 - r, y2, x1 + r, y2,
           x1, y2, x1, y2 - r, x1, y1 + r, x1, y1]
    return cv.create_polygon(pts, smooth=True, **kw)


class RoundButton(tk.Canvas):

    def __init__(self, parent, text, command, *, width, height, px,
                 bg, fg, hover_bg, hover_fg=None, border=None,
                 parent_bg, font, radius=10):
        super().__init__(parent, width=width, height=height, bg=parent_bg,
                         highlightthickness=0, bd=0, cursor="hand2")
        self._cmd = command
        self._text = text
        self._cw, self._ch, self._r = width, height, radius
        self._colors = dict(bg=bg, fg=fg, hbg=hover_bg,
                            hfg=hover_fg or fg, border=border)
        self._font = font
        self._parent_bg = parent_bg
        self._t = 0.0
        self._target = 0.0
        self._job = None
        self._enabled = True
        self._draw()
        self.bind("<Enter>", lambda e: self._chover(1.0))
        self.bind("<Leave>", lambda e: self._chover(0.0))
        self.bind("<ButtonRelease-1>", self._click)

    def _draw(self):
        self.delete("all")
        c = self._colors
        if self._enabled:
            fill = _mix(c["bg"], c["hbg"], _ease(self._t))
            fg = _mix(c["fg"], c["hfg"], _ease(self._t))
            outline = c["border"] or fill
        else:
            pb = self._parent_bg
            light = sum(int(pb[i:i + 2], 16) for i in (1, 3, 5)) > 600
            if light:
                fill, fg = "#eceef2", "#a8afbb"
                outline = "#e1e4ea"
            else:
                fill = _mix(pb, WHITE, 0.10)
                fg = _mix(pb, WHITE, 0.42)
                outline = _mix(pb, WHITE, 0.22) if c["border"] else fill
        _round_rect(self, 1, 1, self._cw - 1, self._ch - 1, self._r,
                    fill=fill, outline=outline, width=1)
        self.create_text(self._cw / 2, self._ch / 2, text=self._text,
                         fill=fg, font=self._font)

    def _hover(self, target):
        if not self._enabled:
            return
        self._target = target
        if self._job is None:
            self._animate()

    def _animate(self):
        step = 0.16 if self._target > self._t else -0.16
        self._t = min(1.0, max(0.0, self._t + step))
        self._draw()
        if abs(self._t - self._target) > 1e-6:
            self._job = self.after(16, self._animate)
        else:
            self._job = None

    def _click(self, e):
        if self._enabled and 0 <= e.x <= self._cw and 0 <= e.y <= self._ch:
            self._cmd()

    def set_text(self, text):
        self._text = text
        self._draw()

    def set_enabled(self, on):
        self._enabled = on
        self.config(cursor="hand2" if on else "arrow")
        if not on:
            self._t = self._target = 0.0
        self._draw()


class ActivityBar(tk.Canvas):

    def __init__(self, parent, height, bg, track, fill):
        super().__init__(parent, height=height, bg=bg,
                         highlightthickness=0, bd=0)
        self._track, self._fill, self._h = track, fill, height
        self._pos, self._running, self._job = 0.0, False, None
        self._done_color = None
        self.bind("<Configure>", lambda e: self._draw())

    def _draw(self):
        self.delete("all")
        w = self.winfo_width() or 1
        h = self._h
        _round_rect(self, 0, 0, w, h, h / 2, fill=self._track, outline="")
        if self._done_color:
            _round_rect(self, 0, 0, w, h, h / 2, fill=self._done_color,
                        outline="")
            return
        if not self._running:
            return
        seg = max(w * 0.3, 80)
        x = _ease(self._pos) * (w + seg) - seg
        x1, x2 = max(0, x), min(w, x + seg)
        if x2 > x1:
            _round_rect(self, x1, 0, x2, h, h / 2, fill=self._fill,
                        outline="")

    def start(self):
        self._done_color = None
        if not self._running:
            self._running, self._pos = True, 0.0
            self._tick()

    def _tick(self):
        self._pos = (self._pos + 0.011) % 1.0
        self._draw()
        self._job = self.after(16, self._tick)

    def stop(self, color=None):
        self._running = False
        if self._job is not None:
            self.after_cancel(self._job)
            self._job = None
        self._done_color = color
        self._draw()


class Spinner(tk.Canvas):

    def __init__(self, parent, size, bg, color):
        super().__init__(parent, width=size, height=size, bg=bg,
                         highlightthickness=0, bd=0)
        self._size, self._color = size, color
        self._angle, self._job = 0, None

    def start(self):
        if self._job is None:
            self._tick()

    def _tick(self):
        self.delete("all")
        s, w = self._size, max(2, self._size // 8)
        box = (w, w, s - w, s - w)
        self.create_oval(*box, outline=RED_TINT, width=w)
        self.create_arc(*box, start=self._angle, extent=100, style="arc",
                        outline=self._color, width=w)
        self._angle = (self._angle - 12) % 360
        self._job = self.after(24, self._tick)

    def stop(self):
        if self._job is not None:
            self.after_cancel(self._job)
            self._job = None
        self.delete("all")


class StatusDot(tk.Canvas):

    def __init__(self, parent, size, bg):
        super().__init__(parent, width=size, height=size, bg=bg,
                         highlightthickness=0, bd=0)
        self._s, self._bg = size, bg
        self._state, self._phase, self._job = "idle", 0.0, None
        self._draw()

    def set_bg(self, bg):
        self._bg = bg
        self.config(bg=bg)
        self._draw()

    def set_state(self, state):
        self._state = state
        if state == "running" and self._job is None:
            self._pulse()
        elif state != "running" and self._job is not None:
            self.after_cancel(self._job)
            self._job = None
        self._draw()

    def _pulse(self):
        self._phase = (self._phase + 0.06) % 1.0
        self._draw()
        self._job = self.after(30, self._pulse)

    def _draw(self):
        self.delete("all")
        s = self._s
        c = s / 2
        if self._state == "running":
            import math
            k = (math.sin(self._phase * 2 * math.pi) + 1) / 2
            r = s * (0.28 + 0.2 * k)
            self.create_oval(c - r, c - r, c + r, c + r,
                             fill=_mix(RED_SOFT, RED_TINT, k), outline="")
            r2 = s * 0.2
            self.create_oval(c - r2, c - r2, c + r2, c + r2,
                             fill=RED, outline="")
            return
        color = {"idle": LINE_STRONG, "ok": OK, "fail": RED}[self._state]
        r = s * 0.24
        self.create_oval(c - r, c - r, c + r, c + r, fill=color, outline="")

    def _load_months(self, select=None, reset=True):
        found = month_folders()
        self.downloads = scan_downloads(REPORTS)
        self.choices = []
        if found:
            latest, latest_path = found[0]
            nxt = add_months(latest, 1)
            self.choices.append((f"New  ·  {month_title(nxt)}",
                                 nxt, True, latest_path))
        for d, f in found:

        self.pick.config(values=[c[0] for c in self.choices])

            self._refresh(reset)

            idx = next((i for i, c in enumerate(self.choices)
                        if c[1] == select and not c[2]), 0)
        else:
            idx = 0 if self.downloads else (1 if len(self.choices) > 1 else 0)
        self._choose(idx, reset)

        self.target, self.is_new, self.source = d, new, src
        self._refresh(reset)

        self.focus_set()

    def _show_results(self):
        if not self.results:
            return
        title = {"run": "Automation", "qc": "QC", "network": "Network copy",
                 "email": "Emails"}.get(self.last_kind, "Results")
        lines = []
        for i, r in enumerate(REPORTS):
            state, text = self.results.get(i, ("idle", "not run"))
            mark = "✓" if state == "ok" else ("✕" if state == "fail" else "·")
            lines.append(f"{mark}  {r['name']}  —  {text}")
        for i, r in enumerate(REPORTS):
            if self.results.get(i, ("",))[0] == "fail":
                errs = [l.strip() for l in self.details[i]
                        if "ERROR" in l or l.strip().startswith(("- ", "diff"))]
                if errs:
                    lines.append("")
                    lines.append(f"{r['name']}:")
                    lines += ["   " + e for e in errs[:8]]
        bad = any(v[0] == "fail" for v in self.results.values())
        (messagebox.showerror if bad else messagebox.showinfo)(
            title, "\n".join(lines))

    def _start_run(self):

            return

        missing = [r["name"] for i, r in enumerate(REPORTS)

            if missing:
                messagebox.showerror(
                    "WebI files missing",
                    "These WebI files are not in Downloads:\n\n"
                    + "\n".join(missing)

            msg = (f"Create {self.folder.name} from {self.source.name}?\n\n"

                   f"• Move {len(self.downloads)} WebI files from Downloads "

    def _tick(self):
        if not self.running:
            return
        m, s = divmod(int(time.time() - self._t0), 60)
        self.elapsed.config(text=f"{m:d}:{s:02d}")
        self._timer = self.after(500, self._tick)

    # ---------------- worker (background thread) ----------------

    def _worker(self, kind, target, is_new, source, downloads):
        put = self.q.put
        folder = month_folder_path(target)
        failed = set()
        ok = bad = 0

        def rlog(i):
            return lambda m: put(("line", (i, str(m))))

        def step(i, state, note=None):
            put(("step", (i, state, note)))

        def row(i, state, text, color=None):
            put(("row", (i, state, text, color)))

        try:
            if kind == "run":
                if is_new:
                    step(0, "running")
                    put(("status", f"Copying {source.name} → {folder.name}"))
                    create_month_folder(source, folder, log=lambda m: None)
                    step(0, "ok", folder.name)
                    step(1, "running")
                    put(("status", "Removing old WebI files"))
                    n = clear_webi(folder, log=lambda m: None)
                    step(1, "ok", f"{n} removed")

                phases = [
                    (2, "Moving WebI", "WebI placed",
                     lambda i, r, f: place_webi(r, f, downloads.get(i),
                                                log=rlog(i))),
                    (3, "Updating", "Updated",
                     lambda i, r, f: update_report(r, f, log=rlog(i))),
                    (4, "Renaming", "Done",
                     lambda i, r, f: rename_files(r, f, target, log=rlog(i))),
                ]
                for s_i, verb, done_text, fn in phases:
                    step(s_i, "running")
                    before = len(failed)
                    for i, r in enumerate(REPORTS):
                        if i in failed:
                            continue
                        put(("status", f"{verb} {i + 1}/{len(REPORTS)} · "
                                       f"{r['name']}"))
                        row(i, "running", verb + "…", RED)
                        try:
                            fn(i, r, report_folder(r, folder))
                            row(i, "running" if s_i < 4 else "ok",
                                done_text, OK if s_i == 4 else MUTED)
                        except Exception as e:
                            failed.add(i)
                            rlog(i)(f"ERROR: {e}")
                            row(i, "fail", str(e), RED)
                    n_bad = len(failed) - before
                    step(s_i, "fail" if n_bad else "ok",
                         f"{n_bad} failed" if n_bad else None)
                bad = len(failed)
                ok = len(REPORTS) - bad

            elif kind == "qc":
                for i, r in enumerate(REPORTS):
                    put(("status", f"QC {i + 1}/{len(REPORTS)} · {r['name']}"))
                    row(i, "running", "Checking…", RED)
                    try:
                        passed = qc_report_monthly(r, report_folder(r, folder),
                                                   log=rlog(i))
                    except Exception as e:
                        rlog(i)(f"ERROR: {e}")
                        passed = False
                    if passed:
                        ok += 1
                        row(i, "ok", "QC passed", OK)
                    else:
                        bad += 1
                        row(i, "fail", "QC failed · click for details", RED)

            elif kind == "network":
                for i, r in enumerate(REPORTS):
                    put(("status", f"Copying {i + 1}/{len(REPORTS)} · "
                                   f"{r['name']}"))
                    row(i, "running", "Copying…", RED)
                    try:
                        dest = copy_to_network(r, report_folder(r, folder),
                                               target, log=rlog(i))
                        ok += 1
                        row(i, "ok", "→ " + str(dest), OK)
                    except Exception as e:
                        bad += 1
                        rlog(i)(f"ERROR: {e}")
                        row(i, "fail", str(e), RED)

            elif kind == "email":
                put(("status", "Creating email drafts"))
                lines = []
                n = draft_emails(REPORTS, target, log=lines.append)
                ok = n
                put(("note", "\n".join(lines)))

        except Exception as e:
            bad += 1
            put(("fatal", str(e)))

        put(("done", (kind, ok, bad)))

    # ---------------- queue -> UI ----------------

    def _drain(self):
        while True:
            try:
                kind, data = self.q.get_nowait()
            except queue.Empty:
                break
            if kind == "status":
                self.status.config(text=self._clip(str(data), 52), fg=TEXT)
            elif kind == "line":
                i, text = data
                self.details[i].append(text)
            elif kind == "row":
                i, state, text, color = data
                if state in ("ok", "fail"):
                    self.results[i] = (state, text)
                elif self.results.get(i, ("",))[0] != "fail":
                    self.results[i] = ("ok", text)
            elif kind == "step":
                i, state, note = data
                if state == "running":
                    self.phase.config(
                        text=f"Step {i + 1} of {len(STEPS)}  ·  {STEPS[i]}")
            elif kind == "fatal":
                messagebox.showerror("Stopped", str(data))
            elif kind == "note":
                self._last_note = data
            elif kind == "done":
                self._finish(*data)
        self.after(80, self._drain)

    def _finish(self, kind, ok, bad):
        self.running = False
        self._stop_badge()
        self.spinner.stop()
        self.bar.stop(color=RED if bad else OK)
        if self._timer is not None:
            self.after_cancel(self._timer)
            self._timer = None
        self._busy_btn.set_text(self._orig_text)

        if kind == "run":
            summary = f"{ok} updated, {bad} failed"
            badge = "FAILED" if bad else "DONE"
            target = self.target
            self._load_months(select=target, reset=False)
        elif kind == "qc":
            summary = f"QC: {ok} passed, {bad} failed"
            badge = "QC FAILED" if bad else "QC PASSED"
        elif kind == "network":
            summary = f"{ok} copied to network, {bad} failed"
            badge = "FAILED" if bad else "COPIED"
        else:
            summary = (f"{ok} email draft(s) opened" if not bad
                       else "Could not create emails")
            badge = "FAILED" if bad else "EMAILS"

        self._gate()
        self.phase.config(text="")
        self.status.config(text=summary, fg=RED if bad else OK)
        if self.results:
            self.details_lbl.config(text="View details",
                                    fg=RED if bad else SUBTLE)
        self._set_badge(badge, WHITE, RED_DEEP if bad else OK)
        self._toast_msg(summary, not bad)
