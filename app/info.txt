CREATE DATABASE MedicoDB;
GO

USE MedicoDB;
GO

CREATE TABLE Departments (
    DepartmentID INT PRIMARY KEY,
    DepartmentName VARCHAR(100)
);

CREATE TABLE Doctors (
    DoctorID INT PRIMARY KEY,
    DoctorName VARCHAR(100),
    Specialization VARCHAR(100),
    DepartmentID INT,
    Salary DECIMAL(10,2)
);

CREATE TABLE Patients (
    PatientID INT PRIMARY KEY,
    PatientName VARCHAR(100),
    Age INT,
    Gender VARCHAR(10),
    City VARCHAR(50)
);

CREATE TABLE Appointments (
    AppointmentID INT PRIMARY KEY,
    PatientID INT,
    DoctorID INT,
    AppointmentDate DATE,
    Status VARCHAR(30),
    Fee DECIMAL(10,2)
);

SELECT *
FROM Patients
WHERE Age > 40;

SELECT *
FROM Doctors
ORDER BY Salary DESC;

SELECT COUNT(*) AS TotalPatients
FROM Patients;

SELECT AVG(Salary) AS AverageSalary
FROM Doctors;

SELECT Status, COUNT(*) AS TotalAppointments
FROM Appointments
GROUP BY Status;

SELECT
    D.DoctorName,
    DP.DepartmentName
FROM Doctors D
INNER JOIN Departments DP
ON D.DepartmentID = DP.DepartmentID;

SELECT
    P.PatientName,
    D.DoctorName,
    A.AppointmentDate,
    A.Status,
    A.Fee
FROM Appointments A
INNER JOIN Patients P
ON A.PatientID = P.PatientID
INNER JOIN Doctors D
ON A.DoctorID = D.DoctorID;

SELECT
    DoctorID,
    SUM(Fee) AS TotalRevenue
FROM Appointments
WHERE Status = 'Completed'
GROUP BY DoctorID;

WITH DoctorRanking AS (
    SELECT
        DoctorName,
        Salary,
        DENSE_RANK() OVER (
            ORDER BY Salary DESC
        ) AS Ranking
    FROM Doctors
)
SELECT * FROM DoctorRanking;

CREATE VIEW AppointmentDetails AS
SELECT
    A.AppointmentID,
    P.PatientName,
    D.DoctorName,
    A.AppointmentDate,
    A.Status,
    A.Fee
FROM Appointments A
INNER JOIN Patients P
ON A.PatientID = P.PatientID
INNER JOIN Doctors D
ON A.DoctorID = D.DoctorID;
GO

SELECT * FROM AppointmentDetails;
