# table_finder.py
# Finds every separate table in every sheet of an Excel file.
# Requires: pip install openpyxl pandas

import openpyxl
import pandas as pd
from collections import deque


def load_grid(ws):
    """Read a sheet into a 2D list, filling merged cells with their top-left value."""
    max_r, max_c = ws.max_row, ws.max_column
    grid = [[ws.cell(r, c).value for c in range(1, max_c + 1)] for r in range(1, max_r + 1)]
    for rng in ws.merged_cells.ranges:
        val = ws.cell(rng.min_row, rng.min_col).value
        for r in range(rng.min_row, rng.max_row + 1):
            for c in range(rng.min_col, rng.max_col + 1):
                grid[r - 1][c - 1] = val
    return grid


def is_filled(v):
    return v is not None and str(v).strip() != ""


def find_blocks(grid, gap=1):
    """Group filled cells into blocks. Cells within `gap` empty cells of each other join the same block."""
    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    seen = [[False] * cols for _ in range(rows)]
    blocks = []

    for r in range(rows):
        for c in range(cols):
            if seen[r][c] or not is_filled(grid[r][c]):
                continue
            q = deque([(r, c)])
            seen[r][c] = True
            r0 = r1 = r
            c0 = c1 = c
            while q:
                cr, cc = q.popleft()
                r0, r1 = min(r0, cr), max(r1, cr)
                c0, c1 = min(c0, cc), max(c1, cc)
                for dr in range(-gap, gap + 1):
                    for dc in range(-gap, gap + 1):
                        nr, nc = cr + dr, cc + dc
                        if 0 <= nr < rows and 0 <= nc < cols and not seen[nr][nc] and is_filled(grid[nr][nc]):
                            seen[nr][nc] = True
                            q.append((nr, nc))
            blocks.append((r0, r1, c0, c1))
    return blocks


def detect_header(block_rows, scan=5):
    """Header = first row (within the first few) that is mostly text and followed by a row with numbers."""
    def text_ratio(row):
        filled = [v for v in row if is_filled(v)]
        if not filled:
            return 0
        return sum(isinstance(v, str) for v in filled) / len(filled)

    for i in range(min(scan, len(block_rows) - 1)):
        if text_ratio(block_rows[i]) >= 0.7 and text_ratio(block_rows[i + 1]) < 0.7:
            return i
    return 0


def make_unique(headers):
    out, count = [], {}
    for i, h in enumerate(headers):
        name = str(h).strip() if is_filled(h) else f"col_{i + 1}"
        if name in count:
            count[name] += 1
            name = f"{name}_{count[name]}"
        else:
            count[name] = 0
        out.append(name)
    return out


def extract_tables(path, min_rows=2, min_cols=2):
    """Return a list of dicts: sheet, location, title, and a DataFrame for each table found."""
    wb = openpyxl.load_workbook(path, data_only=True)
    results = []

    for ws in wb.worksheets:
        grid = load_grid(ws)
        if not grid:
            continue
        for (r0, r1, c0, c1) in find_blocks(grid):
            block = [row[c0:c1 + 1] for row in grid[r0:r1 + 1]]
            # drop fully empty rows/cols inside the block
            block = [row for row in block if any(is_filled(v) for v in row)]
            if len(block) < min_rows:
                continue
            keep = [j for j in range(len(block[0])) if any(is_filled(row[j]) for row in block)]
            block = [[row[j] for j in keep] for row in block]
            if len(keep) < min_cols:
                continue

            h = detect_header(block)
            title_rows = block[:h]
            title = " ".join(str(v) for row in title_rows for v in row if is_filled(v)) or None

            df = pd.DataFrame(block[h + 1:], columns=make_unique(block[h]))
            df = df.dropna(how="all")

            results.append({
                "sheet": ws.title,
                "location": f"R{r0 + 1}C{c0 + 1}:R{r1 + 1}C{c1 + 1}",
                "title": title,
                "data": df,
            })
    return results


if __name__ == "__main__":
    import sys
    file_path = sys.argv[1] if len(sys.argv) > 1 else "sample.xlsx"
    for t in extract_tables(file_path):
        print(f"\n=== Sheet: {t['sheet']} | {t['location']} | Title: {t['title']}")
        print(t["data"].head())
